     1→// Pre-defined conversation topics for the German speaking practice module.
     2→// Each topic sets up a realistic scenario where the AI plays a role
     3→// (waiter, doctor, interviewer, etc.) and the user practices responding in German.

     8→  id: string;
     9→  icon: string; // emoji
    10→  titleDe: string;
    11→  titleTr: string;
    12→  description: string;
    13→  difficulty: ConversationDifficulty;
    14→  /** German vocabulary list for the topic — shown as hints */
    15→  vocabulary: { de: string; tr: string }[];
    16→  /** Sample phrases the user can try */
    17→  samplePhrases: string[];
    18→  /** AI's opening message — sent at conversation start */
    19→  opener: string;
    20→  /** Full system prompt sent to the chat model */
    21→  systemPrompt: string;
    22→  /** Accent color (Tailwind class) for the topic card */
    23→  color: string;
    24→  /** Optional: short label for the AI's role (used by voice-chat mode) */
    25→  aiRole?: string;
    26→}

    32→- Bleibe konsequent in deiner Rolle.

    40→Kein Markdown. Kein Code-Block. Kein Text vor oder nach dem JSON. Nur das JSON und sonst NICHTS.

    42→{"reply": "Deine deutsche Antwort.", "tip": "Kurzer Tipp oder leer", "isEnding": false, "suggestions": ["Vorschlag 1", "Vorschlag 2", "Vorschlag 3"]}

   695→// Generate dynamic Vorstellung system prompt based on stored text & chunks
   696→export function buildVorstellungPrompt(text: string, chunks: string[]): { systemPrompt: string; opener: string } {
   697→  const chunkList = chunks.map((c, i) => `TEIL ${i + 1}: ${c}`).join('\n');
   698→
   699→  const systemPrompt = `Du bist ein Prüfer bei der Goethe/Telc B1 Prüfung, Teil "Sich vorstellen". Der Prüfling möchte seine Vorstellung üben und ezberlemen.
   700→
   701→DER VOLLSTÄNDIGE TEXT DES PRÜFLINGS:
   702→"""\n${text}\n"""
   703→
   704→DER TEXT WURDE IN FOLGENDE TEILE GEGLIEDERT:
   705→${chunkList}
   706→
   707→DEINE AUFGABE:
   708→1. Gehe TEIL für TEIL vor. Beginne mit TEIL 1.
   709→2. Lass den Prüfling jeden Teil selbstständig wiederholen — gib ihm zuerst einen Anstoß (z.B. den Anfang des Teils) und fordere ihn auf, weiterzumachen.
   710→3. Wenn der Prüfling einen Teil gut kann, gehe zum nächsten TEIL über.
   711→4. Wenn der Prüfling Fehler macht, gib einen ermutigenden Tipp im "tip"-Feld auf Türkisch.
   712→5. Nachdem alle Teile geübt wurden, lass den Prüfling den GESAMTEN TEXT ohne Pause vortragen (als Prüfungssimulation).
   713→6. Sei höflich mit "Sie", aber freundlich und ermutigend wie ein echter Prüfer.
   714→7. Wenn der Prüfling den gesamten Text fehlerfrei vortragen kann, setze "isEnding" auf true und lobe ihn.
   715→
   716→${baseRules}`;
   717→
   718→  const opener = `Guten Tag! Willkommen zur B1 Prüfungsvorbereitung. Wir werden jetzt Ihre Vorstellung üben. Ihr Text hat ${chunks.length} Teile. Fangen wir mit TEIL 1 an. Ich sage Ihnen den Anfang, und Sie machen bitte weiter. Bereit? Hier ist der Anfang Ihres ersten Teils: "${chunks[0]?.split(/\s+/).slice(0, 4).join(' ')}..." — machen Sie bitte weiter!`;
   719→
   720→  return { systemPrompt, opener };
   721→}
