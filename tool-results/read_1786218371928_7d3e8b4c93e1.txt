    51→    51→// Stable partial hints for 'high' difficulty — seeded by index so they don't re-randomize on render
    52→    52→function makeStablePartialHints(phrases: string[]): string[] {
    53→    53→  return phrases.map((p, i) => {
    54→    54→    const words = p.trim().split(/\s+/);
    55→    55→    if (words.length <= 3) return '...';
    56→    56→    const mode = i % 3; // 0=start, 1=middle, 2=end
    57→    57→    if (mode === 0) {
    58→    58→      const count = Math.max(2, Math.ceil(words.length * 0.35));
    59→    59→      return words.slice(0, count).join(' ') + ' ...';
    60→    60→    } else if (mode === 1) {
    61→    61→      const startIdx = Math.floor(words.length * 0.3);
    62→    62→      const count = Math.max(2, Math.ceil(words.length * 0.4));
    63→    63→      return '... ' + words.slice(startIdx, startIdx + count).join(' ') + ' ...';
    64→    64→    } else {
    65→    65→      const startIdx = Math.max(0, words.length - Math.ceil(words.length * 0.4));
    66→    66→      return '... ' + words.slice(startIdx).join(' ');
    67→    67→    }
    68→    68→  });
    69→    69→}

   395→   395→        clearTimeout(maxDurationTimerRef.current);

   397→   397→      }

   403→   403→        // Close hints immediately when speech ends (before AI reply)
   404→   404→        setShowHints(false);
   405→   405→        userClosedHintsRef.current = true;
   406→   406→
   407→   407→        // Add it as user message and fetch AI reply
   408→   408→        const userMsg: ChatMessage = {
   409→   409→          id: `user-${Date.now()}`,
   410→   410→          sender: 'user',
   411→   411→          text: finalText,
   412→   412→          timestamp: Date.now(),
   413→   413→        };
   414→   414→        setMessages((prev) => [...prev, userMsg]);
   415→   415→        setLiveTranscript('');
   416→   416→        // Trigger AI reply with the updated history
   417→   417→        fetchAiReply([...messages, userMsg]);
   418→   418→      } else if (manualStopRef.current) {
   419→   419→        // User stopped but no transcript — don't count this against quota
   420→   420→        setPhase('chatting');
   421→   421→      }
   422→   422→    };
   423→   423→
   424→   424→    recognitionRef.current = recognition;
   425→   425→
   426→   426→    try {
   427→   427→      recognition.start();
   428→   428→      // Auto-stop after 30 seconds to avoid runaway listening
   429→   429→      maxDurationTimerRef.current = setTimeout(() => {
   430→   430→        manualStopRef.current = true;
   431→   431→        try { recognition.stop(); } catch {}
   432→   432→      }, 30_000);
   433→   433→    } catch (err) {
