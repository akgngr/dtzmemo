    50→/** Simple heuristic: does a string contain Turkish-only special chars? */
    51→function hasTurkishSpecialChars(str: string): boolean {
    52→  return /[şığüşçöİŞĞÜŞÇÖ]/.test(str);
    53→}

    55→/** Check if a string looks like a CSV header (e.g. "german,turkish" or "Almanca,Türkçe") */
    56→function looksLikeHeader(first: string, second: string): boolean {
    57→  const headerKeywords = ['german', 'turkish', 'almanca', 'türkçe', 'deutsch', 'de', 'tr', 'kelime', 'word', 'çeviri', 'translation'];
    58→  const combined = (first + ' ' + second).toLowerCase();
    59→  return headerKeywords.some((k) => combined.includes(k));
    60→}

    62→/** Parse CSV content into word pairs */
    63→function parseCSV(content: string): CustomWord[] {
    64→  // Strip BOM if present
    65→  let cleaned = content.replace(/^\uFEFF/, '');
    66→
    67→  // Normalize line endings
    68→  cleaned = cleaned.replace(/\r\n/g, '\n').replace(/\r/g, '\n');
    69→
    70→  const lines = cleaned.split('\n').filter((line) => line.trim().length > 0);
    71→
    72→  if (lines.length === 0) return [];
    73→
    74→  let startIndex = 0;
    75→
    76→  // Check if the first line is a header
    77→  if (lines.length > 1) {
    78→    const firstLineParts = lines[0].split(',').map((s) => s.trim());
    79→    if (firstLineParts.length >= 2) {
    80→      const col1 = firstLineParts[0];
    81→      const col2 = firstLineParts[1];
    82→      // Skip if it looks like a header
    83→      if (looksLikeHeader(col1, col2) || hasTurkishSpecialChars(col1)) {
    84→        startIndex = 1;
    85→      }
    86→    }
    87→  }
    88→
    89→  const words: CustomWord[] = [];
    90→
    91→  for (let i = startIndex; i < lines.length; i++) {
    92→    const parts = lines[i].split(',').map((s) => s.trim());
    93→    if (parts.length >= 2 && parts[0] && parts[1]) {
    94→      words.push({ german: parts[0], turkish: parts[1] });
    95→    }
    96→  }
    97→
    98→  return words;
    99→}

   101→/** Encode a list to a base64 share string */
   102→function encodeList(list: Omit<CustomWordList, 'id' | 'createdAt' | 'category'>): string {
   103→  const payload = { n: list.name, d: list.description, w: list.words };
   104→  return btoa(JSON.stringify(payload));
   105→}
   106→
   107→/** Decode a base64 share string */
   108→function decodeList(encoded: string): { name: string; description: string; words: CustomWord[] } | null {
   109→  try {
   110→    const payload = JSON.parse(atob(encoded));
   111→    if (payload && payload.w && Array.isArray(payload.w)) {
   112→      return {
   113→        name: payload.n || 'İçe Aktarılan Liste',
   114→        description: payload.d || '',
   115→        words: payload.w.filter(
   116→          (w: any) => typeof w.german === 'string' && typeof w.turkish === 'string' && w.german && w.turkish
   117→        ),
   118→      };
   119→    }
   120→    return null;
   121→  } catch {
   122→    return null;
   123→  }
   124→}
   125→
   126→function generateShareURL(list: Omit<CustomWordList, 'id' | 'createdAt' | 'category'>): string {
   127→  const base64 = encodeList(list);
   128→  return window.location.origin + window.location.pathname + '?import=' + encodeURIComponent(base64);
   129→}

   131→async function copyToClipboard(text: string): Promise<boolean> {
   132→  try {
   133→    await navigator.clipboard.writeText(text);
   134→    return true;
   135→  } catch {
   136→    // Fallback for older browsers / restricted contexts
   137→    try {
   138→      const textarea = document.createElement('textarea');
   139→      textarea.value = text;
   140→      textarea.style.position = 'fixed';
   141→      textarea.style.opacity = '0';
   142→      document.body.appendChild(textarea);
   143→      textarea.select();
   144→      document.execCommand('copy');
   145→      document.body.removeChild(textarea);
   146→      return true;
   147→    } catch {
   148→      return false;
   149→    }
   150→  }
   151→}

   157→function formatDate(isoString: string): string {
   158→  try {
   159→    const d = new Date(isoString);
   160→    return d.toLocaleDateString('tr-TR', { day: 'numeric', month: 'short', year: 'numeric' });
   161→  } catch {
   162→    return isoString;
   163→  }
   164→}
