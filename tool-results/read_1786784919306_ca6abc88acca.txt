    51→    51→// Stable partial hints for 'high' difficulty — seeded by index so they don't re-randomize on render
    52→    52→function makeStablePartialHints(phrases: string[]): string[] {
    53→    53→  return phrases.map((p, i) => {
    54→    54→    const words = p.trim().split(/\s+/);
    55→    55→    if (words.length <= 3) return '...';
    56→    56→    const mode = i % 3; // 0=start, 1=middle, 2=end
    57→    57→    if (mode === 0) {
    58→    58→      const count = Math.max(2, Math.ceil(words.length * 0.35));
    59→    59→      return words.slice(0, count).join(' ') + ' ...';
    60→    60→    } else if (mode === 1) {
    61→    61→      const startIdx = Math.floor(words.length * 0.3);
    62→    62→      const count = Math.max(2, Math.ceil(words.length * 0.4));
    63→    63→      return '... ' + words.slice(startIdx, startIdx + count).join(' ') + ' ...';
    64→    64→    } else {
    65→    65→      const startIdx = Math.max(0, words.length - Math.ceil(words.length * 0.4));
    66→    66→      return '... ' + words.slice(startIdx).join(' ');
    67→    67→    }
    68→    68→  });
    69→    69→}
