     1→    55→    if (words.length <= 3) return '...';
     2→    56→    const mode = i % 3; // 0=start, 1=middle, 2=end
     3→    57→    if (mode === 0) {
     4→    58→      const count = Math.max(2, Math.ceil(words.length * 0.35));
     5→    59→      return words.slice(0, count).join(' ') + ' ...';
     6→    60→    } else if (mode === 1) {
     7→    61→      const startIdx = Math.floor(words.length * 0.3);
     8→    62→      const count = Math.max(2, Math.ceil(words.length * 0.4));
     9→    63→      return '... ' + words.slice(startIdx, startIdx + count).join(' ') + ' ...';
    10→    64→    } else {
    11→    65→      const startIdx = Math.max(0, words.length - Math.ceil(words.length * 0.4));
    12→    66→      return '... ' + words.slice(startIdx).join(' ');
    13→    67→    }
    14→    68→  });
    15→    69→}
