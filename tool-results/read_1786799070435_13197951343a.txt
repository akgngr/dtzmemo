    60→    60→/* ═══ Speech Recognition Types ═══ */
    61→    61→interface SRResult { 0: { transcript: string }; isFinal: boolean; }
    62→    62→interface SREvent { results: { length: number; [i: number]: SRResult }; resultIndex: number; }
    63→    63→interface SR {
    64→    64→  lang: string; continuous: boolean; interimResults: boolean; maxAlternatives: number;
    65→    65→  start: () => void; stop: () => void; abort: () => void;
    66→    66→  onresult: ((e: SREvent) => void) | null;

    68→    68→  onend: (() => void) | null;

    73→    73→export function ExamPictureModule() {
    74→    74→  const tts = useTTS({ lang: 'de-DE' });
    75→    75→  const { apiKeys, hasSpeechQuota, incrementSpeechUsage, remainingSpeechQuota, daysUntilSpeechReset, speechUsage } = useAppStore();
    76→    76→
    77→    77→  const [phase, setPhase] = useState<Phase>('select-picture');
    78→    78→  const [selectedPicture, setSelectedPicture] = useState<PicturePrompt | null>(null);
    79→    79→  const [level, setLevel] = useState<Level | null>(null);
    80→    80→  const [messages, setMessages] = useState<ChatMessage[]>([]);
    81→    81→  const [imageDescription, setImageDescription] = useState('');
    82→    82→  const [textInput, setTextInput] = useState('');
    83→    83→  const [showHints, setShowHints] = useState(false);
    84→    84→  const [error, setError] = useState<string | null>(null);
    85→    85→  const [fallbackNotice, setFallbackNotice] = useState(false);
    86→    86→  const [liveTranscript, setLiveTranscript] = useState('');
    87→    87→  const [browserSupported, setBrowserSupported] = useState(true);
    88→    88→
    89→    89→  const messagesEndRef = useRef<HTMLDivElement | null>(null);
    90→    90→  const recognitionRef = useRef<SR | null>(null);
    91→    91→  const finalTranscriptRef = useRef('');
    92→    92→  const manualStopRef = useRef(false);
    93→    93→  const silenceTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    94→    94→  const maxDurTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
    95→    95→  const lastAiRef = useRef<ChatMessage | null>(null);
    96→    96→  const levelRef = useRef<Level | null>(null);
    97→    97→  const msgsRef = useRef<ChatMessage[]>([]);
    98→    98→  const imgDescRef = useRef('');
    99→    99→
   100→   100→  useEffect(() => { levelRef.current = level; }, [level]);
   101→   101→  useEffect(() => { msgsRef.current = messages; }, [messages]);
   102→   102→  useEffect(() => { imgDescRef.current = imageDescription; }, [imageDescription]);
   103→   103→
   104→   104→  const quotaRemaining = remainingSpeechQuota();
   105→   105→  const quotaExhausted = !hasSpeechQuota();
   106→   106→
   107→   107→  // Scroll
   108→   108→  useEffect(() => { messagesEndRef.current?.scrollIntoView({ behavior: 'smooth', block: 'end' }); }, [messages, phase, liveTranscript]);
   109→   109→
   110→   110→  // Auto-play AI
   111→   111→  useEffect(() => {
   112→   112→    const last = messages[messages.length - 1];
   113→   113→    if (!last || last.sender !== 'ai' || last === lastAiRef.current) return;
   114→   114→    lastAiRef.current = last;
   115→   115→    tts.speak(last.text).catch(() => {});
   116→   116→  }, [messages, tts]);
   117→   117→
   118→   118→  // Browser check
   119→   119→  useEffect(() => {
   120→   120→    if (typeof window !== 'undefined') {
   121→   121→      const w = window as any;
   122→   122→      if (!w.SpeechRecognition && !w.webkitSpeechRecognition) setBrowserSupported(false);
   123→   123→    }
   124→   124→  }, []);
   125→   125→
   126→   126→  /* ═══ API Call ═══ */
   127→   127→  const fetchAiReply = useCallback(async (history: ChatMessage[], isFirst: boolean) => {
   128→   128→    setPhase('loading-ai');
   129→   129→    setError(null);
   130→   130→    try {
   131→   131→      const apiMessages = history
   132→   132→        .filter(m => m.id !== 'opener')
   133→   133→        .map(m => ({ role: m.sender === 'user' ? 'user' : 'assistant', content: m.text }));
   134→   134→
   135→   135→      const res = await fetch('/api/exam-picture-chat', {
   136→   136→        method: 'POST',
   137→   137→        headers: { 'Content-Type': 'application/json' },
   138→   138→        body: JSON.stringify({
   139→   139→          imageUrl: selectedPicture!.imageUrl,
   140→   140→          messages: apiMessages,
   141→   141→          level: levelRef.current!,
   142→   142→          imageDescription: imgDescRef.current || undefined,
   143→   143→          isFirstMessage: isFirst,
   144→   144→          zhipuKey: apiKeys.zhipuKey || undefined,
   145→   145→          openaiKey: apiKeys.openaiKey || undefined,
   146→   146→          claudeKey: apiKeys.claudeKey || undefined,
   147→   147→          googleAiKey: apiKeys.googleAiKey || undefined,
   148→   148→        }),
   149→   149→      });
   150→   150→      if (!res.ok) {
   151→   151→        const d = await res.json().catch(() => ({}));
   152→   152→        throw new Error(d.error || `HTTP ${res.status}`);
   153→   153→      }
   154→   154→      const data = await res.json();
   155→   155→
   156→   156→      if (data.imageDescription) setImageDescription(data.imageDescription);
   157→   157→
   158→   158→      const aiMsg: ChatMessage = {
   159→   159→        id: `ai-${Date.now()}`,
   160→   160→        sender: 'ai',
   161→   161→        text: data.reply || '...',
   162→   162→        tip: data.tip || '',
   163→   163→        isEnding: Boolean(data.isEnding),
   164→   164→        suggestions: Array.isArray(data.suggestions) && data.suggestions.length > 0 ? data.suggestions : undefined,
   165→   165→      };
   166→   166→      setMessages(prev => [...prev, aiMsg]);
   167→   167→
   168→   168→      if (aiMsg.isEnding) setTimeout(() => setPhase('ended'), 1500);
   169→   169→      else setPhase('chatting');
   170→   170→    } catch (err) {
   171→   171→      setError((err as Error).message);
   172→   172→      setPhase('chatting');
   173→   173→    }
   174→   174→  }, [selectedPicture, apiKeys]);
   175→   175→
   176→   176→  /* ═══ Start Session ═══ */
   177→   177→  const startSession = useCallback(async (pic: PicturePrompt, lvl: Level) => {
   178→   178→    setSelectedPicture(pic);
   179→   179→    setLevel(lvl);
   180→   180→    setMessages([]);
   181→   181→    setImageDescription('');
   182→   182→    setPhase('analyzing');
   183→   183→    setError(null);
   184→   184→
   185→   185→    // First API call: analyze image + get opening message
   186→   186→    try {
   187→   187→      const res = await fetch('/api/exam-picture-chat', {
   188→   188→        method: 'POST',
   189→   189→        headers: { 'Content-Type': 'application/json' },
   190→   190→        body: JSON.stringify({
   191→   191→          imageUrl: pic.imageUrl,
   192→   192→          messages: [],
   193→   193→          level: lvl,
   194→   194→          isFirstMessage: true,
   195→   195→          fallbackDescription: pic.description,
   196→   196→          zhipuKey: apiKeys.zhipuKey || undefined,
   197→   197→          openaiKey: apiKeys.openaiKey || undefined,
   198→   198→          claudeKey: apiKeys.claudeKey || undefined,
   199→   199→          googleAiKey: apiKeys.googleAiKey || undefined,
   200→   200→        }),
   201→   201→      });
   202→   202→      if (!res.ok) {
   203→   203→        const d = await res.json().catch(() => ({}));
   204→   204→        throw new Error(d.error || `HTTP ${res.status}`);
   205→   205→      }
   206→   206→      const data = await res.json();
   207→   207→      if (data.imageDescription) setImageDescription(data.imageDescription);
   208→   208→      if (data.usedFallback) setFallbackNotice(true);
   209→   209→      else setFallbackNotice(false);
   210→   210→
   211→   211→      const opener: ChatMessage = {
   212→   212→        id: 'opener',
   213→   213→        sender: 'ai',
   214→   214→        text: data.reply || 'Bitte beschreiben Sie dieses Bild.',
   215→   215→        tip: data.tip || '',
   216→   216→        suggestions: Array.isArray(data.suggestions) && data.suggestions.length > 0 ? data.suggestions : undefined,
   217→   217→      };
   218→   218→      setMessages([opener]);
   219→   219→      setPhase('chatting');
   220→   220→    } catch (err) {
   221→   221→      console.error('[ExamPicture] startSession failed:', err);
   222→   222→      setError((err as Error).message);
   223→   223→      setPhase('select-level');
   224→   224→    }
   225→   225→  }, [apiKeys]);
   226→   226→
   227→   227→  /* ═══ Send User Message ═══ */
   228→   228→  const sendMessage = useCallback(async (text: string) => {
   229→   229→    if (!text.trim() || phase !== 'chatting') return;
   230→   230→    setTextInput('');
   231→   231→    setShowHints(false);
   232→   232→    const userMsg: ChatMessage = { id: `user-${Date.now()}`, sender: 'user', text: text.trim(), timestamp: Date.now() };
   233→   233→    const updated = [...msgsRef.current, userMsg];
   234→   234→    setMessages(updated);
   235→   235→    await fetchAiReply(updated, false);
   236→   236→  }, [phase, fetchAiReply]);
   237→   237→
   238→   238→  /* ═══ Speech Recognition ═══ */
   239→   239→  const startListening = useCallback(() => {
   240→   240→    if (quotaExhausted) { setError(`Bu ayki ses tanıma kotan doldu. ${daysUntilSpeechReset()} gün sonra yenilenecek.`); return; }
   241→   241→    tts.stop(); setError(null);
   242→   242→    setLiveTranscript(''); finalTranscriptRef.current = ''; manualStopRef.current = false;
   243→   243→    if (recognitionRef.current) try { recognitionRef.current.abort(); } catch {}
   244→   244→
   245→   245→    const w = window as any;
   246→   246→    const ctor = (w.SpeechRecognition || w.webkitSpeechRecognition) as SRCtor | undefined;
   247→   247→    if (!ctor) { setBrowserSupported(false); return; }
   248→   248→
   249→   249→    const rec = new ctor();
   250→   250→    rec.lang = 'de-DE'; rec.continuous = true; rec.interimResults = true; rec.maxAlternatives = 1;
   251→   251→    rec.onstart = () => setPhase('listening');
   252→   252→    rec.onresult = (e: SREvent) => {
   253→   253→      if (silenceTimerRef.current) { clearTimeout(silenceTimerRef.current); silenceTimerRef.current = null; }
   254→   254→      silenceTimerRef.current = setTimeout(() => { manualStopRef.current = true; try { rec.stop(); } catch {} }, 5000);
   255→   255→      let interim = '', final_ = finalTranscriptRef.current;
   256→   256→      for (let i = e.resultIndex; i < e.results.length; i++) {
   257→   257→        if (e.results[i].isFinal) final_ += e.results[i][0].transcript + ' ';
   258→   258→        else interim += e.results[i][0].transcript;
   259→   259→      }
   260→   260→      finalTranscriptRef.current = final_;
   261→   261→      setLiveTranscript((final_ + interim).trim());
   262→   262→    };
   263→   263→    rec.onerror = (e: any) => {
   264→   264→      if (e.error === 'not-allowed') setError('Mikrofon izni reddedildi.');
   265→   265→      else if (e.error === 'no-speech' && !finalTranscriptRef.current.trim()) setError('Ses algılanamadı.');
   266→   266→      else if (e.error !== 'aborted') setError(`Tanıma hatası: ${e.error}`);
   267→   267→      setPhase('chatting');
   268→   268→    };
   269→   269→    rec.onend = () => {
   270→   270→      if (maxDurTimerRef.current) { clearTimeout(maxDurTimerRef.current); maxDurTimerRef.current = null; }
   271→   271→      if (silenceTimerRef.current) { clearTimeout(silenceTimerRef.current); silenceTimerRef.current = null; }
   272→   272→      const t = finalTranscriptRef.current.trim();
   273→   273→      if (t) { incrementSpeechUsage(); sendMessage(t); }
   274→   274→      else if (manualStopRef.current) setPhase('chatting');
   275→   275→    };
   276→   276→    recognitionRef.current = rec;
   277→   277→    try {
   278→   278→      rec.start();
   279→   279→      maxDurTimerRef.current = setTimeout(() => { manualStopRef.current = true; try { rec.stop(); } catch {} }, 30_000);
   280→   280→    } catch { setError('Dinleme başlatılamadı.'); setPhase('chatting'); }
   281→   281→  }, [tts, quotaExhausted, daysUntilSpeechReset, incrementSpeechUsage, sendMessage]);
   282→   282→
   283→   283→  const stopListening = useCallback(() => {
   284→   284→    manualStopRef.current = true;
   285→   285→    if (maxDurTimerRef.current) { clearTimeout(maxDurTimerRef.current); maxDurTimerRef.current = null; }
   286→   286→    if (silenceTimerRef.current) { clearTimeout(silenceTimerRef.current); silenceTimerRef.current = null; }
   287→   287→    if (recognitionRef.current) try { recognitionRef.current.stop(); } catch {}
   288→   288→  }, []);
   289→   289→
   290→   290→  const handleMicToggle = useCallback(() => {
   291→   291→    if (phase === 'listening') stopListening();
   292→   292→    else if (phase === 'chatting') startListening();
   293→   293→  }, [phase, startListening, stopListening]);
   294→   294→
   295→   295→  const reset = useCallback(() => {
   296→   296→    if (recognitionRef.current) try { recognitionRef.current.abort(); } catch {}
   297→   297→    tts.stop();
   298→   298→    setPhase(selectedPicture ? 'select-level' : 'select-picture');
   299→   299→    setMessages([]); setImageDescription(''); setLevel(null); setError(null);
   300→   300→  }, [selectedPicture, tts]);
   301→   301→
   302→   302→  const lastAiMsg = messages[messages.length - 1];
   303→   303→  const lastSuggestions = lastAiMsg?.sender === 'ai' && lastAiMsg.suggestions?.length ? lastAiMsg.suggestions : undefined;
   304→   304→
   305→   305→  /* ═════════════ RENDER ═════════════ */
   306→   306→
   307→   307→  // ─── Phase 1: Select Picture ───
   308→   308→  if (phase === 'select-picture') {
   309→   309→    return (
   310→   310→      <div className="space-y-4">
   311→   311→        <div className="rounded-2xl bg-gradient-to-br from-violet-50 to-purple-50 p-5 border border-violet-100">
   312→   312→          <div className="flex items-center gap-3 mb-3">
   313→   313→            <div className="flex h-10 w-10 items-center justify-center rounded-xl bg-violet-100">
   314→   314→              <Eye className="h-5 w-5 text-violet-600" />
   315→   315→            </div>
   316→   316→            <div>
   317→   317→              <h2 className="text-lg font-bold text-violet-900">Bildbeschreibung</h2>
   318→   318→              <p className="text-xs text-violet-500">Resim Tanımlama - AI Destekli</p>
   319→   319→            </div>
   320→   320→          </div>
   321→   321→          <p className="text-sm text-violet-700 leading-relaxed">
   322→   322→            Bir resim seçin. AI resmi analiz edip size uygun seviyede rehberlik edecek.
   323→   323→            Her seviyede farklı bir deneyim yaşayacaksınız!
   324→   324→          </p>
   325→   325→        </div>
   326→   326→        <div className="grid gap-3">
   327→   327→          {picturePrompts.map((p, i) => (
   328→   328→            <motion.button
   329→   329→              key={p.id}
   330→   330→              initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} transition={{ delay: i * 0.05 }}
   331→   331→              onClick={() => { setSelectedPicture(p); setPhase('select-level'); }}
   332→   332→              className="flex items-center gap-3 rounded-xl border border-gray-200 bg-white p-4 text-left hover:border-violet-300 hover:shadow-md transition-all duration-200"
   333→   333→            >
   334→   334→              <div className="flex h-16 w-24 shrink-0 items-center justify-center rounded-lg overflow-hidden bg-gray-100">
   335→   335→                <img src={p.imageUrl} alt={p.title} className="h-full w-full object-cover" />
   336→   336→              </div>
   337→   337→              <div className="flex-1 min-w-0">
   338→   338→                <p className="text-sm font-semibold text-gray-900">{p.title}</p>
   339→   339→                <p className="text-xs text-gray-500">{p.titleTr}</p>
   340→   340→                <p className="text-xs text-gray-400 mt-1 line-clamp-2">{p.descriptionTr}</p>
   341→   341→              </div>
   342→   342→              <ChevronRight className="h-4 w-4 text-gray-400 shrink-0" />
   343→   343→            </motion.button>
   344→   344→          ))}
   345→   345→        </div>
   346→   346→      </div>
   347→   347→    );
   348→   348→  }
   349→   349→
   350→   350→  // ─── Phase 2: Select Level ───
   351→   351→  if (phase === 'select-level' && selectedPicture) {
   352→   352→    return (
   353→   353→      <div className="space-y-4">
   354→   354→        <button onClick={() => setPhase('select-picture')} className="flex items-center gap-1.5 text-sm text-gray-500 hover:text-gray-800 transition-colors">
   355→   355→          <ArrowLeft className="h-4 w-4" /> Resim Seçimine Dön
   356→   356→        </button>
   357→   357→        <div className="rounded-2xl overflow-hidden border border-gray-200">
   358→   358→          <img src={selectedPicture.imageUrl} alt={selectedPicture.title} className="w-full h-48 object-cover" />
   359→   359→          <div className="p-4 bg-white">
   360→   360→            <h3 className="font-bold text-gray-900">{selectedPicture.title}</h3>
   361→   361→            <p className="text-xs text-gray-500 mt-1">{selectedPicture.descriptionTr}</p>
   362→   362→          </div>
   363→   363→        </div>
   364→   364→        <div className="text-center py-2">
   365→   365→          <h3 className="text-lg font-bold text-gray-800">Seviyeni Seç</h3>
   366→   366→          <p className="text-sm text-gray-500">AI nasıl rehberlik etsin?</p>
   367→   367→        </div>
   368→   368→        <div className="space-y-3">
   369→   369→          {LEVELS.map((lv, i) => (
   370→   370→            <motion.button
   371→   371→              key={lv.id} initial={{ opacity: 0, y: 10 }} animate={{ opacity: 1, y: 0 }} transition={{ delay: i * 0.08 }}
   372→   372→              whileHover={{ scale: 1.01 }} whileTap={{ scale: 0.98 }}
   373→   373→              onClick={() => startSession(selectedPicture, lv.id)}
   374→   374→              className={cn('w-full rounded-2xl border-2 p-4 text-left hover:shadow-md transition-all', lv.borderColor, 'bg-white hover:border-current')}
   375→   375→            >
   376→   376→              <div className="flex items-center gap-3">
   377→   377→                <div className={cn('shrink-0 w-12 h-12 rounded-xl flex items-center justify-center text-2xl', lv.bgColor)}>
   378→   378→                  {lv.icon}
   379→   379→                </div>
   380→   380→                <div className="flex-1 min-w-0">
   381→   381→                  <div className="flex items-center gap-2">
   382→   382→                    <span className={cn('font-bold', lv.color)}>{lv.label}</span>
   383→   383→                    <span className={cn('text-xs px-1.5 py-0.5 rounded-full', lv.bgColor, lv.color)}>{lv.labelTr}</span>
   384→   384→                  </div>
   385→   385→                  <p className="text-xs text-gray-500 mt-1">{lv.description}</p>
   386→   386→                </div>
   387→   387→              </div>
   388→   388→            </motion.button>
   389→   389→          ))}
   390→   390→        </div>
   391→   391→      </div>
   392→   392→    );
   393→   393→  }
