    35→interface QuizQuestion {
    36→  word: WordPair;
    37→  options: string[];
    38→  correctIndex: number;
    39→}
    40→
    41→function shuffleArray<T>(array: T[]): T[] {
    42→  const shuffled = [...array];
    43→  for (let i = shuffled.length - 1; i > 0; i--) {
    44→    const j = Math.floor(Math.random() * (i + 1));
    45→    [shuffled[i], shuffled[j]] = [shuffled[j], shuffled[i]];
    46→  }
    47→  return shuffled;
    48→}
    49→
    50→function generateQuestions(
    51→  pool: WordPair[],
    52→  count: number,
    53→  direction: Direction
    54→): QuizQuestion[] {
    55→  const shuffled = shuffleArray(pool);
    56→  const selected = shuffled.slice(0, Math.min(count, shuffled.length));
    57→
    58→  return selected.map((word) => {
    59→    const otherWords = pool.filter(
    60→      (w) =>
    61→        w.id !== word.id &&
    62→        (direction === 'de-to-tr'
    63→          ? !selected.slice(0, selected.indexOf(word) >= 0 ? selected.indexOf(word) : 0)
    64→              .some((s) => s.turkish === w.turkish)
    65→          : true)
    66→    );
    67→
    68→    const field: 'turkish' | 'german' = direction === 'de-to-tr' ? 'turkish' : 'german';
    69→    const correctValue = word[field];
    70→
    71→    // Pick 3 unique wrong answers
    72→    const usedValues = new Set<string>([correctValue]);
    73→    const wrongOptions: string[] = [];
    74→    const candidates = shuffleArray(otherWords);
    75→
    76→    for (const candidate of candidates) {
    77→      const val = candidate[field];
    78→      if (!usedValues.has(val)) {
    79→        wrongOptions.push(val);
    80→        usedValues.add(val);
    81→      }
    82→      if (wrongOptions.length >= 3) break;
    83→    }
    84→
    85→    // Fallback: if we don't have enough, allow duplicates (shouldn't happen with a decent pool)
    86→    while (wrongOptions.length < 3 && pool.length > 1) {
    87→      const rand = pool[Math.floor(Math.random() * pool.length)];
    88→      const val = rand[field];
    89→      if (val !== correctValue) {
    90→        wrongOptions.push(val);
    91→      }
    92→    }
    93→
    94→    const options = shuffleArray([correctValue, ...wrongOptions]);
    95→    const correctIndex = options.indexOf(correctValue);
    96→
    97→    return { word, options, correctIndex };
    98→  });
    99→}
