    74→function shuffle<T>(arr: T[]): T[] {
    75→  const a = [...arr];
    76→  for (let i = a.length - 1; i > 0; i--) {
    77→    const j = Math.floor(Math.random() * (i + 1));
    78→    [a[i], a[j]] = [a[j], a[i]];
    79→  }
    80→  return a;
    81→}

    83→function tokenize(s: string): string[] {
    84→  return s
    85→    .toLowerCase()
    86→    .split(/[^\p{L}\p{N}]+/u)
    87→    .filter(Boolean);
    88→}

    90→// Jaccard similarity on Turkish tokens, with a small length-similarity bonus.
    91→function similarity(a: WordPair, b: WordPair): number {
    92→  const ta = new Set(tokenize(a.turkish));
    93→  const tb = new Set(tokenize(b.turkish));
    94→  if (ta.size === 0 || tb.size === 0) return 0;
    95→  let shared = 0;
    96→  ta.forEach((t) => {
    97→    if (tb.has(t)) shared += 1;
    98→  });
    99→  const jaccard = shared / (ta.size + tb.size - shared);
   100→  const lengthPenalty = Math.abs(a.turkish.length - b.turkish.length) / 100;
   101→  return jaccard - lengthPenalty;
   102→}

   104→function buildPool(selectedCategories: string[]): WordPair[] {
   105→  let pool = [...wordPairs];
   106→  if (selectedCategories.length > 0) {
   107→    pool = pool.filter((w) => selectedCategories.includes(w.category));
   108→  }
   109→  return shuffle(pool).slice(0, SESSION_SIZE);
   110→}
   111→
   112→function generateOptions(correct: WordPair, mode: DifficultyMode): WordPair[] {
   113→  const pool = wordPairs.filter((w) => w.id !== correct.id && w.turkish !== correct.turkish);
   114→  let candidates: WordPair[] = [];
   115→
   116→  if (mode === 'easy') {
   117→    // Same category as correct
   118→    candidates = pool.filter((w) => w.category === correct.category);
   119→    // Fallback to random others if same category has fewer than 3
   120→    if (candidates.length < 3) {
   121→      const others = shuffle(pool.filter((w) => w.category !== correct.category));
   122→      candidates = [...candidates, ...others];
   123→    }
   124→  } else if (mode === 'medium') {
   125→    // Different categories than correct
   126→    candidates = pool.filter((w) => w.category !== correct.category);
   127→  } else {
   128→    // Hard: pick most similar (shared Turkish words + similar length)
   129→    const scored = pool.map((w) => ({ w, score: similarity(correct, w) }));
   130→    scored.sort((a, b) => b.score - a.score);
   131→    candidates = scored.slice(0, Math.min(10, scored.length)).map((s) => s.w);
   132→  }
   133→
   134→  // Pick 3 unique-by-Turkish distractors
   135→  const distractors: WordPair[] = [];
   136→  const seen = new Set<string>();
   137→  for (const w of shuffle(candidates)) {
   138→    if (!seen.has(w.turkish)) {
   139→      distractors.push(w);
   140→      seen.add(w.turkish);
   141→      if (distractors.length >= 3) break;
   142→    }
   143→  }
   144→
   145→  return shuffle([correct, ...distractors]);
   146→}
