    41→function shuffleArray<T>(array: T[]): T[] {
    42→  const shuffled = [...array];
    43→  for (let i = shuffled.length - 1; i > 0; i--) {
    44→    const j = Math.floor(Math.random() * (i + 1));
    45→    [shuffled[i], shuffled[j]] = [shuffled[j], shuffled[i]];
    46→  }
    47→  return shuffled;
    48→}
    49→
    50→function generateQuestions(
    51→  pool: WordPair[],
    52→  count: number,
    53→  direction: Direction
    54→): QuizQuestion[] {
    55→  const shuffled = shuffleArray(pool);
    56→  const selected = shuffled.slice(0, Math.min(count, shuffled.length));
    57→
    58→  return selected.map((word) => {
    59→    const otherWords = pool.filter((w) => w.id !== word.id);
    60→    const field: 'turkish' | 'german' = direction === 'de-to-tr' ? 'turkish' : 'german';
    61→    const correctValue = word[field];
    62→
    63→    const usedValues = new Set<string>([correctValue]);
    64→    const wrongOptions: string[] = [];
    65→    const candidates = shuffleArray(otherWords);
    66→
    67→    for (const candidate of candidates) {
    68→      const val = candidate[field];
    69→      if (!usedValues.has(val)) {
    70→        wrongOptions.push(val);
    71→        usedValues.add(val);
    72→      }
    73→      if (wrongOptions.length >= 3) break;
    74→    }
    75→
    76→    while (wrongOptions.length < 3 && pool.length > 1) {
    77→      const rand = pool[Math.floor(Math.random() * pool.length)];
    78→      const val = rand[field];
    79→      if (val !== correctValue && !wrongOptions.includes(val)) {
    80→        wrongOptions.push(val);
    81→      }
    82→    }
    83→
    84→    const options = shuffleArray([correctValue, ...wrongOptions]);
    85→    const correctIndex = options.indexOf(correctValue);
    86→
    87→    return { word, options, correctIndex };
    88→  });
    89→}
