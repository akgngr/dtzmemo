    56→/** Simple heuristic: does a string contain Turkish-only special chars? */
    57→function hasTurkishSpecialChars(str: string): boolean {
    58→  return /[şığüşçöİŞĞÜŞÇÖ]/.test(str);
    59→}

    61→/** Check if a string looks like a CSV header (e.g. "german,turkish" or "Almanca,Türkçe") */
    62→function looksLikeHeader(first: string, second: string): boolean {
    63→  const headerKeywords = ['german', 'turkish', 'almanca', 'türkçe', 'deutsch', 'de', 'tr', 'kelime', 'word', 'çeviri', 'translation'];
    64→  const combined = (first + ' ' + second).toLowerCase();
    65→  return headerKeywords.some((k) => combined.includes(k));
    66→}

    68→/** Parse CSV content into word pairs */
    69→function parseCSV(content: string): CustomWord[] {
    70→  // Strip BOM if present
    71→  let cleaned = content.replace(/^\uFEFF/, '');
    72→
    73→  // Normalize line endings
    74→  cleaned = cleaned.replace(/\r\n/g, '\n').replace(/\r/g, '\n');
    75→
    76→  const lines = cleaned.split('\n').filter((line) => line.trim().length > 0);
    77→
    78→  if (lines.length === 0) return [];
    79→
    80→  let startIndex = 0;
    81→
    82→  // Check if the first line is a header
    83→  if (lines.length > 1) {
    84→    const firstLineParts = lines[0].split(',').map((s) => s.trim());
    85→    if (firstLineParts.length >= 2) {
    86→      const col1 = firstLineParts[0];
    87→      const col2 = firstLineParts[1];
    88→      // Skip if it looks like a header
    89→      if (looksLikeHeader(col1, col2) || hasTurkishSpecialChars(col1)) {
    90→        startIndex = 1;
    91→      }
    92→    }
    93→  }
    94→
    95→  const words: CustomWord[] = [];
    96→
    97→  for (let i = startIndex; i < lines.length; i++) {
    98→    const parts = lines[i].split(',').map((s) => s.trim());
    99→    if (parts.length >= 2 && parts[0] && parts[1]) {
   100→      words.push({ german: parts[0], turkish: parts[1] });
   101→    }
   102→  }
   103→
   104→  return words;
   105→}

   107→/** Encode a list to a base64 share string */
   108→function encodeList(list: Omit<CustomWordList, 'id' | 'createdAt' | 'category'>): string {
   109→  const payload = { n: list.name, d: list.description, w: list.words };
   110→  return btoa(JSON.stringify(payload));
   111→}
   112→
   113→/** Decode a base64 share string */
   114→function decodeList(encoded: string): { name: string; description: string; words: CustomWord[] } | null {
   115→  try {
   116→    const payload = JSON.parse(atob(encoded));
   117→    if (payload && payload.w && Array.isArray(payload.w)) {
   118→      return {
   119→        name: payload.n || 'İçe Aktarılan Liste',
   120→        description: payload.d || '',
   121→        words: payload.w.filter(
   122→          (w: any) => typeof w.german === 'string' && typeof w.turkish === 'string' && w.german && w.turkish
   123→        ),
   124→      };
   125→    }
   126→    return null;
   127→  } catch {
   128→    return null;
   129→  }
   130→}

   132→function generateShareURL(list: Omit<CustomWordList, 'id' | 'createdAt' | 'category'>): string {
   133→  const base64 = encodeList(list);
   134→  return window.location.origin + window.location.pathname + '?import=' + encodeURIComponent(base64);
   135→}

   137→async function copyToClipboard(text: string): Promise<boolean> {
   138→  try {
   139→    await navigator.clipboard.writeText(text);
   140→    return true;
   141→  } catch {
   142→    // Fallback for older browsers / restricted contexts
   143→    try {
   144→      const textarea = document.createElement('textarea');
   145→      textarea.value = text;
   146→      textarea.style.position = 'fixed';
   147→      textarea.style.opacity = '0';
   148→      document.body.appendChild(textarea);
   149→      textarea.select();
   150→      document.execCommand('copy');
   151→      document.body.removeChild(textarea);
   152→      return true;
   153→    } catch {
   154→      return false;
   155→    }
   156→  }
   157→}

   163→/** Build CSV string from word pairs */
   164→function buildCSV(words: CustomWord[]): string {
   165→  const lines = ['almanca,türkçe'];
   166→  for (const w of words) {
   167→    const german = w.german.includes(',') ? `"${w.german}"` : w.german;
   168→    const turkish = w.turkish.includes(',') ? `"${w.turkish}"` : w.turkish;
   169→    lines.push(`${german},${turkish}`);
   170→  }
   171→  return lines.join('\n');
   172→}

   174→/** Trigger a CSV file download */
   175→function downloadCSV(content: string, filename: string) {
   176→  const blob = new Blob([content], { type: 'text/csv;charset=utf-8;' });
   177→  const url = URL.createObjectURL(blob);
   178→  const a = document.createElement('a');
   179→  a.href = url;
   180→  a.download = filename;
   181→  document.body.appendChild(a);
   182→  a.click();
   183→  document.body.removeChild(a);
   184→  URL.revokeObjectURL(url);
   185→}

   223→function formatDate(isoString: string): string {
   224→  try {
   225→    const d = new Date(isoString);
