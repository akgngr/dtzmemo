     1→'use client';
     2→
     3→import React, { useState, useRef, useEffect, useCallback } from 'react';
     4→import { motion, AnimatePresence } from 'framer-motion';
     5→import {
     6→  Mic, Square, Send, Volume2, Lightbulb, X,
     7→  Loader2, MessageCircle,
     8→} from 'lucide-react';
     9→import { Card, CardContent } from '@/components/ui/card';
    10→import { Button } from '@/components/ui/button';
    11→import { Badge } from '@/components/ui/badge';
    12→import { type ConversationTopic, buildVorstellungPrompt } from '@/lib/conversation-topics';
    13→import { useTTS } from '@/hooks/use-tts';
    14→import { useAppStore } from '@/lib/store';
    15→import { cn } from '@/lib/utils';
    16→
    17→type Sender = 'user' | 'ai';
    18→
    19→interface ChatMessage {
    20→  id: string;
    21→  sender: Sender;
    22→  text: string;
    23→  tip?: string;
    24→  isEnding?: boolean;
    25→  suggestions?: string[];
    26→  timestamp: number;
    27→}
    28→
    29→interface ConversationChatProps {
    30→  topic: ConversationTopic;
    31→  onExit: (summary: ConversationSummary) => void;
    32→}
    33→
    34→export interface ConversationSummary {
    35→  topicId: string;
    36→  topicTitleDe: string;
    37→  topicTitleTr: string;
    38→  messageCount: number;
    39→  userMessageCount: number;
    40→  tipsCount: number;
    41→  durationSec: number;
    42→  messages: ChatMessage[];
    43→}
    44→
    45→type Phase = 'chatting' | 'listening' | 'loading-ai' | 'ended';
    46→
    47→const MAX_TURNS = 10;
    48→
    49→// Minimal typings for the Web Speech API (Google's free STT in Chrome)
    50→interface SpeechRecognitionResultLike {
    51→  0: { transcript: string };
    52→  isFinal: boolean;
    53→}
    54→interface SpeechRecognitionEventLike {
    55→  results: { length: number; [index: number]: SpeechRecognitionResultLike };
    56→  resultIndex: number;
    57→}
    58→interface SpeechRecognitionLike {
    59→  lang: string;
    60→  continuous: boolean;
    61→  interimResults: boolean;
    62→  maxAlternatives: number;
    63→  start: () => void;
    64→  stop: () => void;
    65→  abort: () => void;
    66→  onresult: ((e: SpeechRecognitionEventLike) => void) | null;
    67→  onerror: ((e: any) => void) | null;
    68→  onend: (() => void) | null;
    69→  onstart: (() => void) | null;
    70→}
    71→type SpeechRecognitionCtor = new () => SpeechRecognitionLike;
    72→
    73→export function ConversationChat({ topic, onExit }: ConversationChatProps) {
    74→  // Handle special B1 Vorstellung topic — build dynamic prompt from stored text
    75→  const vorstellungText = useAppStore((s) => s.vorstellungText);
    76→  const vorstellungChunks = useAppStore((s) => s.vorstellungChunks);
    77→
    78→  const isVorstellung = topic.id === 'b1-vorstellung';
    79→  const effectiveTopic = (() => {
    80→    if (isVorstellung && vorstellungText.trim() && vorstellungChunks.length > 0) {
    81→      const { systemPrompt, opener } = buildVorstellungPrompt(vorstellungText, vorstellungChunks);
    82→      return { ...topic, systemPrompt, opener };
    83→    }
    84→    return topic;
    85→  })();
    86→
    87→  const [messages, setMessages] = useState<ChatMessage[]>([
    88→    {
    89→      id: 'opener',
    90→      sender: 'ai',
    91→      text: effectiveTopic.opener,
    92→      timestamp: Date.now(),
    93→    },
    94→  ]);
    95→  const [phase, setPhase] = useState<Phase>('chatting');
    96→  const [textInput, setTextInput] = useState('');
    97→  const [showHints, setShowHints] = useState(false);
    98→  const [showVocab, setShowVocab] = useState(false);
    99→  const [error, setError] = useState<string | null>(null);
   100→  const [liveTranscript, setLiveTranscript] = useState<string>('');
   101→  const [browserSupported, setBrowserSupported] = useState<boolean>(true);
   102→  const [showTypingHint, setShowTypingHint] = useState(false);
   103→  const typingHintTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   104→
   105→  // ===== Google Web Speech API quota tracking =====
   106→  const speechUsage = useAppStore((s) => s.speechUsage);
   107→  const incrementSpeechUsage = useAppStore((s) => s.incrementSpeechUsage);
   108→  const hasSpeechQuota = useAppStore((s) => s.hasSpeechQuota);
   109→  const remainingSpeechQuota = useAppStore((s) => s.remainingSpeechQuota);
   110→  const daysUntilSpeechReset = useAppStore((s) => s.daysUntilSpeechReset);
   111→
   112→  // Recompute remaining on every render — store getters read fresh state
   113→  const quotaRemaining = remainingSpeechQuota();
   114→  const quotaExhausted = !hasSpeechQuota();
   115→  const daysUntilReset = daysUntilSpeechReset();
   116→
   117→  const tts = useTTS({ lang: 'de-DE' });
   118→  const messagesEndRef = useRef<HTMLDivElement | null>(null);
   119→  const startTimeRef = useRef<number>(Date.now());
   120→  const autoPlayRef = useRef<boolean>(true);
   121→  const lastAiMessageRef = useRef<ChatMessage | null>(null);
   122→
   123→  // SpeechRecognition refs
   124→  const recognitionRef = useRef<SpeechRecognitionLike | null>(null);
   125→  const finalTranscriptRef = useRef<string>('');
   126→  const manualStopRef = useRef<boolean>(false);
   127→  const maxDurationTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   128→
   129→  // Lazy-init the SpeechRecognition constructor (browser only)
   130→  const getRecognitionCtor = useCallback((): SpeechRecognitionCtor | null => {
   131→    if (typeof window === 'undefined') return null;
   132→    const w = window as any;
   133→    return ((w.SpeechRecognition || w.webkitSpeechRecognition) as
   134→      | SpeechRecognitionCtor
   135→      | undefined) ?? null;
   136→  }, []);
   137→
   138→  // Typing hint: show after 3s of inactivity in 'chatting' phase
   139→  useEffect(() => {
   140→    // Clear any existing timer
   141→    if (typingHintTimerRef.current) {
   142→      clearTimeout(typingHintTimerRef.current);
   143→      typingHintTimerRef.current = null;
   144→    }
   145→    setShowTypingHint(false);
   146→
   147→    if (phase === 'chatting' && messages.length > 0) {
   148→      typingHintTimerRef.current = setTimeout(() => {
   149→        setShowTypingHint(true);
   150→      }, 3000);
   151→    }
   152→
   153→    return () => {
   154→      if (typingHintTimerRef.current) {
   155→        clearTimeout(typingHintTimerRef.current);
   156→        typingHintTimerRef.current = null;
   157→      }
   158→    };
   159→  }, [phase, messages.length]);
   160→
   161→  // Check support on mount
   162→  useEffect(() => {
   163→    const ctor = getRecognitionCtor();
   164→    if (!ctor) {
   165→      setBrowserSupported(false);
   166→    }
   167→  }, [getRecognitionCtor]);
   168→
   169→  // Auto-scroll to bottom on new message
   170→  useEffect(() => {
   171→    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth', block: 'end' });
   172→  }, [messages, phase, liveTranscript]);
   173→
   174→  // Auto-play AI messages via browser TTS (Google's free voices in Chrome)
   175→  useEffect(() => {
   176→    if (!autoPlayRef.current) return;
   177→    const lastMsg = messages[messages.length - 1];
   178→    if (!lastMsg || lastMsg.sender !== 'ai') return;
   179→    if (lastMsg === lastAiMessageRef.current) return;
   180→    lastAiMessageRef.current = lastMsg;
   181→
   182→    // Browser-only TTS (SpeechSynthesis — Google's free Web Speech API)
   183→    tts.speak(lastMsg.text).catch((e) =>
   184→      console.warn('Browser TTS failed:', e)
   185→    );
   186→  }, [messages, tts]);
   187→
   188→  const fetchAiReply = useCallback(async (history: ChatMessage[]) => {
   189→    setPhase('loading-ai');
   190→    setError(null);
   191→    try {
   192→      const apiMessages = history
   193→        .filter((m) => m.id !== 'opener')
   194→        .map((m) => ({
   195→          role: m.sender === 'user' ? 'user' : 'assistant',
   196→          content: m.text,
   197→        }));
   198→      apiMessages.unshift({ role: 'assistant', content: effectiveTopic.opener });
   199→
   200→      const userTurnIndex = history.filter((m) => m.sender === 'user').length;
   201→
   202→      const { zhipuKey, openaiKey, claudeKey, googleAiKey } = useAppStore.getState().apiKeys;
   203→
   204→      const response = await fetch('/api/chat', {
   205→        method: 'POST',
   206→        headers: { 'Content-Type': 'application/json' },
   207→        body: JSON.stringify({
   208→          systemPrompt:
   209→            effectiveTopic.systemPrompt +
   210→            `\n\nBu konuşma ${MAX_TURNS} turdan oluşacak. ${
   211→              userTurnIndex >= MAX_TURNS - 1
   212→                ? 'Bu son tur — vedalaşarak bitir.'
   213→                : ''
   214→            }`,
   215→          messages: apiMessages,
   216→          temperature: 0.7,
   217→          maxTokens: isVorstellung ? 600 : 400,
   218→          zhipuKey: zhipuKey || undefined,
   219→          openaiKey: openaiKey || undefined,
   220→          claudeKey: claudeKey || undefined,
   221→          googleAiKey: googleAiKey || undefined,
   222→        }),
   223→      });
   224→      if (!response.ok) {
   225→        const errData = await response.json().catch(() => ({}));
   226→        throw new Error(errData.error || errData.detail || `HTTP ${response.status}`);
   227→      }
   228→      const data = await response.json() as { reply?: string; tip?: string; isEnding?: boolean; suggestions?: string[] };
   229→      const aiMsg: ChatMessage = {
   230→        id: `ai-${Date.now()}`,
   231→        sender: 'ai',
   232→        text: (data.reply || '').trim() || '...',
   233→        tip: data.tip || '',
   234→        isEnding: Boolean(data.isEnding) || userTurnIndex >= MAX_TURNS - 1,
   235→        suggestions: Array.isArray(data.suggestions) && data.suggestions.length > 0 ? data.suggestions : undefined,
   236→        timestamp: Date.now(),
   237→      };
   238→      setMessages((prev) => [...prev, aiMsg]);
   239→
   240→      if (aiMsg.isEnding) {
   241→        setTimeout(() => setPhase('ended'), 1500);
   242→      } else {
   243→        setPhase('chatting');
   244→      }
   245→    } catch (err) {
   246→      console.error('[ConversationChat] AI reply failed:', err);
   247→      setError('Yanıt alınamadı: ' + (err as Error).message);
   248→      setPhase('chatting');
   249→    }
   250→  }, [effectiveTopic]);
   251→
   252→  // ===== Browser SpeechRecognition (Google's free STT) =====
   253→  const startListening = useCallback(async () => {
   254→    const ctor = getRecognitionCtor();
   255→    if (!ctor) {
   256→      setBrowserSupported(false);
   257→      setError('Tarayıcı ses tanımayı desteklemiyor. Chrome veya Edge kullanın. Yazarak cevap verebilirsiniz.');
   258→      return;
   259→    }
   260→
   261→    // Quota check — if exhausted, refuse to start and tell user when it resets
   262→    if (!hasSpeechQuota()) {
   263→      setError(
   264→        `Bu ayki ücretsiz Google Web Speech API kotan doldu (${speechUsage.monthlyLimit} tanıma). ` +
   265→        `Kota her ayın 1'inde otomatik yenilenir. ${daysUntilReset} gün sonra tekrar kullanabilirsin. ` +
   266→        `Bu süre içinde yazarak cevap vermeye devam edebilirsin.`
   267→      );
   268→      return;
   269→    }
   270→
   271→    // Stop any playing audio before recording
   272→    tts.stop();
   273→    setError(null);
   274→    setLiveTranscript('');
   275→    finalTranscriptRef.current = '';
   276→    manualStopRef.current = false;
   277→
   278→    // Stop any existing recognition
   279→    if (recognitionRef.current) {
   280→      try { recognitionRef.current.abort(); } catch {}
   281→    }
   282→
   283→    const recognition = new ctor();
   284→    recognition.lang = 'de-DE';
   285→    recognition.continuous = true;
   286→    recognition.interimResults = true;
   287→    recognition.maxAlternatives = 1;
   288→
   289→    recognition.onstart = () => {
   290→      setPhase('listening');
   291→    };
   292→
   293→    recognition.onresult = (event: SpeechRecognitionEventLike) => {
   294→      let interim = '';
   295→      let final = finalTranscriptRef.current;
   296→      for (let i = event.resultIndex; i < event.results.length; i++) {
   297→        const result = event.results[i];
   298→        const txt = result[0].transcript;
   299→        if (result.isFinal) {
   300→          final += txt + ' ';
   301→        } else {
   302→          interim += txt;
   303→        }
   304→      }
   305→      finalTranscriptRef.current = final;
   306→      setLiveTranscript((final + interim).trim());
   307→    };
   308→
   309→    recognition.onerror = (e: any) => {
   310→      console.error('[ConversationChat] SpeechRecognition error:', e);
   311→      if (e.error === 'not-allowed' || e.error === 'service-not-allowed') {
   312→        setError('Mikrofon izni reddedildi. Tarayıcı ayarlarından izin verin.');
   313→      } else if (e.error === 'no-speech') {
   314→        if (!finalTranscriptRef.current.trim()) {
   315→          setError('Ses algılanamadı. Tekrar deneyin veya yazarak cevap verin.');
   316→        }
   317→      } else if (e.error === 'network') {
   318→        setError('Ağ hatası. İnternet bağlantınızı kontrol edin (Web Speech API internet gerektirir).');
   319→      } else if (e.error === 'aborted') {
   320→        // Manual abort — not an error
   321→        return;
   322→      } else {
   323→        setError(`Tanıma hatası: ${e.error}`);
   324→      }
   325→      setPhase('chatting');
   326→    };
   327→
   328→    recognition.onend = () => {
   329→      if (maxDurationTimerRef.current) {
   330→        clearTimeout(maxDurationTimerRef.current);
   331→        maxDurationTimerRef.current = null;
   332→      }
   333→      const finalText = finalTranscriptRef.current.trim();
   334→      if (finalText) {
   335→        // Successful STT call — increment monthly counter
   336→        incrementSpeechUsage();
   337→
   338→        // Add it as user message and fetch AI reply
   339→        const userMsg: ChatMessage = {
   340→          id: `user-${Date.now()}`,
   341→          sender: 'user',
   342→          text: finalText,
   343→          timestamp: Date.now(),
   344→        };
   345→        setMessages((prev) => [...prev, userMsg]);
   346→        setLiveTranscript('');
   347→        // Trigger AI reply with the updated history
   348→        fetchAiReply([...messages, userMsg]);
   349→      } else if (manualStopRef.current) {
   350→        // User stopped but no transcript — don't count this against quota
   351→        setPhase('chatting');
   352→      }
   353→    };
   354→
   355→    recognitionRef.current = recognition;
   356→
   357→    try {
   358→      recognition.start();
   359→      // Auto-stop after 30 seconds to avoid runaway listening
   360→      maxDurationTimerRef.current = setTimeout(() => {
   361→        manualStopRef.current = true;
   362→        try { recognition.stop(); } catch {}
   363→      }, 30_000);
   364→    } catch (err) {
   365→      console.error('[ConversationChat] start failed:', err);
   366→      setError('Dinleme başlatılamadı. Yazarak cevap verebilirsiniz.');
   367→      setPhase('chatting');
   368→    }
   369→  }, [getRecognitionCtor, messages, tts, fetchAiReply, hasSpeechQuota, speechUsage.monthlyLimit, daysUntilReset, incrementSpeechUsage]);
   370→
   371→  const stopListening = useCallback(() => {
   372→    manualStopRef.current = true;
   373→    if (maxDurationTimerRef.current) {
   374→      clearTimeout(maxDurationTimerRef.current);
   375→      maxDurationTimerRef.current = null;
   376→    }
   377→    if (recognitionRef.current) {
   378→      try { recognitionRef.current.stop(); } catch {}
   379→    }
   380→  }, []);
   381→
   382→  // Mic toggle
   383→  const handleMicToggle = useCallback(async () => {
   384→    if (phase === 'listening') {
   385→      stopListening();
   386→    } else if (phase === 'chatting') {
   387→      await startListening();
   388→    }
   389→  }, [phase, startListening, stopListening]);
   390→
   391→  const handleSendText = useCallback(async () => {
   392→    const text = textInput.trim();
   393→    if (!text || phase !== 'chatting') return;
   394→    setTextInput('');
   395→    const userMsg: ChatMessage = {
   396→      id: `user-${Date.now()}`,
   397→      sender: 'user',
   398→      text,
   399→      timestamp: Date.now(),
   400→    };
   401→    setMessages((prev) => [...prev, userMsg]);
   402→    await fetchAiReply([...messages, userMsg]);
   403→  }, [textInput, phase, messages, fetchAiReply]);
   404→
   405→  const handleEndConversation = useCallback(() => {
   406→    // Stop any active recognition
   407→    if (recognitionRef.current) {
   408→      try { recognitionRef.current.abort(); } catch {}
   409→    }
   410→    tts.stop();
   411→    setPhase('ended');
   412→  }, [tts]);
   413→
   414→  // Build and send the summary when phase becomes 'ended'
   415→  useEffect(() => {
   416→    if (phase !== 'ended') return;
   417→    const userMessageCount = messages.filter((m) => m.sender === 'user').length;
   418→    const tipsCount = messages.filter((m) => m.tip && m.tip.trim()).length;
   419→    const summary: ConversationSummary = {
   420→      topicId: topic.id,
   421→      topicTitleDe: topic.titleDe,
   422→      topicTitleTr: topic.titleTr,
   423→      messageCount: messages.length,
   424→      userMessageCount,
   425→      tipsCount,
   426→      durationSec: Math.round((Date.now() - startTimeRef.current) / 1000),
   427→      messages,
   428→    };
   429→    onExit(summary);
   430→  }, [phase, messages, topic, onExit]);
   431→
   432→  // Replay an AI message via browser TTS
   433→  const replayMessage = useCallback((msg: ChatMessage) => {
   434→    tts.speak(msg.text);
   435→  }, [tts]);
   436→
   437→  // Get dynamic suggestions from the last AI message, or undefined to fall back to static phrases
   438→  const lastAiMsg = [...messages].reverse().find((m) => m.sender === 'ai' && m.suggestions && m.suggestions.length > 0);
   439→  const lastAiSuggestions = lastAiMsg?.suggestions;
   440→
   441→  const isBusy = phase === 'listening' || phase === 'loading-ai';
   442→
   443→  return (
   444→    <div className="flex flex-col h-[calc(100vh-12rem)] md:h-[calc(100vh-8rem)]">
   445→      {/* Topic header */}
   446→      <div className={cn('rounded-xl p-3 text-white shadow-sm', topic.color)}>
   447→        <div className="flex items-center justify-between gap-2">
   448→          <div className="flex items-center gap-2 min-w-0">
   449→            <span className="text-2xl">{topic.icon}</span>
   450→            <div className="min-w-0">
   451→              <h2 className="text-base font-bold leading-tight truncate">{topic.titleDe}</h2>
   452→              <p className="text-xs opacity-90 truncate">{topic.titleTr} · {topic.difficulty}</p>
   453→            </div>
   454→          </div>
   455→          <div className="flex items-center gap-1 shrink-0">
   456→            <Button
   457→              size="sm"
   458→              variant="ghost"
   459→              className="text-white hover:bg-white/20 h-8 px-2"
   460→              onClick={() => setShowVocab((v) => !v)}
   461→              aria-label="Kelime listesi"
   462→            >
   463→              <Lightbulb className="h-4 w-4" />
   464→            </Button>
   465→            <Button
   466→              size="sm"
   467→              variant="ghost"
   468→              className="text-white hover:bg-white/20 h-8 px-2"
   469→              onClick={() => setShowHints((v) => !v)}
   470→              aria-label="Örnek cümleler"
   471→            >
   472→              <MessageCircle className="h-4 w-4" />
   473→            </Button>
   474→            <Button
   475→              size="sm"
   476→              variant="ghost"
   477→              className="text-white hover:bg-white/20 h-8 px-2"
   478→              onClick={handleEndConversation}
   479→            >
   480→              <X className="h-4 w-4" />
   481→            </Button>
   482→          </div>
   483→        </div>
   484→      </div>
   485→
   486→      {/* Vocabulary drawer (collapsible) */}
   487→      <AnimatePresence>
   488→        {showVocab && (
   489→          <motion.div
   490→            initial={{ opacity: 0, height: 0 }}
   491→            animate={{ opacity: 1, height: 'auto' }}
   492→            exit={{ opacity: 0, height: 0 }}
   493→            className="overflow-hidden"
   494→          >
   495→            <Card className="border-0 shadow-sm mt-2">
   496→              <CardContent className="p-3">
   497→                <div className="flex items-center justify-between mb-2">
   498→                  <span className="text-xs font-semibold uppercase text-muted-foreground">Önemli Kelimeler</span>
   499→                  <button onClick={() => setShowVocab(false)} className="text-xs text-muted-foreground hover:text-foreground">
   500→                    <X className="h-3 w-3" />
   501→                  </button>
   502→                </div>
   503→                <div className="grid grid-cols-2 gap-1.5">
   504→                  {topic.vocabulary.map((v, i) => (
   505→                    <div key={i} className="rounded bg-amber-50 px-2 py-1 text-xs">
   506→                      <span className="font-medium text-amber-900">{v.de}</span>
   507→                      <span className="text-amber-700 ml-1">— {v.tr}</span>
   508→                    </div>
   509→                  ))}
   510→                </div>
   511→              </CardContent>
   512→            </Card>
   513→          </motion.div>
   514→        )}
   515→      </AnimatePresence>
   516→
   517→      {/* Sample phrases drawer — uses dynamic AI suggestions or falls back to static phrases */}
   518→      <AnimatePresence>
   519→        {showHints && (
   520→          <motion.div
   521→            initial={{ opacity: 0, height: 0 }}
   522→            animate={{ opacity: 1, height: 'auto' }}
   523→            exit={{ opacity: 0, height: 0 }}
   524→            className="overflow-hidden"
   525→          >
   526→            <Card className="border-0 shadow-sm mt-2">
   527→              <CardContent className="p-3">
   528→                <div className="flex items-center justify-between mb-2">
   529→                  <span className="text-xs font-semibold uppercase text-muted-foreground">
   530→                    {lastAiSuggestions ? 'Önerilen Cevaplar' : 'Örnek Cümleler'}
   531→                  </span>
   532→                  <button onClick={() => setShowHints(false)} className="text-xs text-muted-foreground hover:text-foreground">
   533→                    <X className="h-3 w-3" />
   534→                  </button>
   535→                </div>
   536→                <div className="flex flex-wrap gap-1.5">
   537→                  {(lastAiSuggestions || topic.samplePhrases).map((p, i) => (
   538→                    <button
   539→                      key={`${p}-${i}`}
   540→                      onClick={() => { setTextInput(p); setShowHints(false); }}
   541→                      className={cn(
   542→                        "rounded-full px-2.5 py-1 text-xs transition-colors",
   543→                        lastAiSuggestions

   551→                </div>

   553→            </Card>
   554→          </motion.div>

   558→      {/* Mode badge — Google free APIs */}

   661→              {/* Dynamic contextual suggestions — shown inline after AI message */}

   683→

   832→          >
