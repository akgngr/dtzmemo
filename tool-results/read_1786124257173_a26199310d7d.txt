    20→function formatMonth(monthStr: string): string {
    21→  const [y, m] = monthStr.split('-').map(Number);
    22→  if (!y || !m) return monthStr;
    23→  return `${MONTH_NAMES_TR[m - 1]} ${y}`;
    24→}

    26→// ===== CSV/JSON Helpers =====
    27→function hasTurkishSpecialChars(str: string): boolean {
    28→  return /[şığüşçöİŞĞÜŞÇÖ]/.test(str);
    29→}
    30→function looksLikeHeader(first: string, second: string): boolean {
    31→  const headerKeywords = ['german', 'turkish', 'almanca', 'türkçe', 'deutsch', 'de', 'tr', 'kelime', 'word', 'çeviri', 'translation'];
    32→  const combined = (first + ' ' + second).toLowerCase();
    33→  return headerKeywords.some((k) => combined.includes(k));
    34→}
    35→function parseCSV(content: string): CustomWord[] {
    36→  let cleaned = content.replace(/^\uFEFF/, '').replace(/\r\n/g, '\n').replace(/\r/g, '\n');
    37→  const lines = cleaned.split('\n').filter((l) => l.trim().length > 0);
    38→  if (lines.length === 0) return [];
    39→  let startIndex = 0;
    40→  if (lines.length > 1) {
    41→    const parts = lines[0].split(',').map((s) => s.trim());
    42→    if (parts.length >= 2 && looksLikeHeader(parts[0], parts[1])) startIndex = 1;
    43→  }
    44→  const words: CustomWord[] = [];
    45→  for (let i = startIndex; i < lines.length; i++) {
    46→    const p = lines[i].split(',').map((s) => s.trim());
    47→    if (p.length >= 2 && p[0] && p[1]) words.push({ german: p[0], turkish: p[1] });
    48→  }
    49→  return words;
    50→}

    52→function buildCSV(list: CustomWordList): string {
    53→  const header = 'Almanca,Türkçe';
    54→  const rows = list.words.map((w) => `${w.german},${w.turkish}`).join('\n');
    55→  return `${header}\n${rows}`;
    56→}

    62→function downloadFile(content: string, filename: string, mimeType: string) {
    63→  const blob = new Blob([content], { type: mimeType });
    64→  const url = URL.createObjectURL(blob);
    65→  const a = document.createElement('a');
    66→  a.href = url; a.download = filename; a.click();
    67→  URL.revokeObjectURL(url);
    68→}

    94→function ApiKeyField({
    95→  label, description, placeholder, value, onChange, testUrl, testMethod = 'POST', testBody, testKeyField, successHint,
    96→}: {
    97→  label: string; description: string; placeholder: string; value: string; onChange: (v: string) => void;
    98→  testUrl: string; testMethod?: string; testBody?: Record<string, unknown>; testKeyField?: string; successHint?: string;
    99→}) {
   100→  const [visible, setVisible] = useState(false);
   101→  const [testing, setTesting] = useState(false);
   102→  const [testResult, setTestResult] = useState<'idle' | 'success' | 'error'>('idle');
   103→
   104→  const handleTest = useCallback(async () => {
   105→    if (!value.trim()) return;
   106→    setTesting(true); setTestResult('idle');
   107→    try {
   108→      const body = testBody ? { ...testBody } : {};
   109→      if (testKeyField) body[testKeyField] = value.trim();
   110→      const res = await fetch(testUrl, { method: testMethod, headers: { 'Content-Type': 'application/json' }, body: Object.keys(body).length > 0 ? JSON.stringify(body) : undefined });
   111→      if (res.status === 401 || res.status === 403) setTestResult('error'); else setTestResult('success');
   112→    } catch { setTestResult('error'); } finally { setTesting(false); }
   113→  }, [value, testUrl, testMethod, testBody, testKeyField]);
   114→
   115→  return (
   116→    <div className="space-y-2">
   117→      <div className="flex items-center justify-between">
   118→        <label className="text-sm font-medium text-foreground">{label}</label>
   119→        {value?.trim() ? (
   120→          <Badge variant="outline" className="border-emerald-200 text-emerald-700 bg-emerald-50 text-[10px]"><CheckCircle2 className="mr-1 h-3 w-3" /> Kayıtlı</Badge>
   121→        ) : (
   122→          <Badge variant="outline" className="border-gray-200 text-gray-500 text-[10px]">Girilmedi</Badge>
   123→        )}
   124→      </div>
   125→      <p className="text-xs text-muted-foreground">{description}</p>
   126→      <div className="flex items-center gap-2">
   127→        <div className="relative flex-1">
   128→          <input type={visible ? 'text' : 'password'} value={value || ''} onChange={(e) => onChange(e.target.value)} className="w-full h-10 rounded-md border border-gray-200 px-3 pr-10 text-sm font-mono focus:border-emerald-400 focus:outline-none" placeholder={placeholder} autoComplete="off" />
   129→          <button type="button" onClick={() => setVisible(!visible)} className="absolute right-2.5 top-1/2 -translate-y-1/2 text-gray-400 hover:text-gray-600">
   130→            {visible ? <EyeOff className="h-4 w-4" /> : <Eye className="h-4 w-4" />}
   131→          </button>
   132→        </div>
   133→        <Button size="sm" variant="outline" onClick={handleTest} disabled={testing || !value?.trim()} className="shrink-0">
   134→          {testing ? 'Test...' : 'Test Et'}
   135→        </Button>
   136→      </div>
   137→      {testResult === 'success' && <p className="text-xs text-emerald-600 flex items-center gap-1"><CheckCircle2 className="h-3 w-3" />{successHint || 'API anahtarı geçerli.'}</p>}
   138→      {testResult === 'error' && <p className="text-xs text-red-600">API anahtarı geçersiz veya erişim reddedildi.</p>}
   139→    </div>
   140→  );
   141→}
