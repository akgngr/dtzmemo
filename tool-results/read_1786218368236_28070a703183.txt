    51→// Stable partial hints for 'high' difficulty — seeded by index so they don't re-randomize on render
    52→function makeStablePartialHints(phrases: string[]): string[] {
    53→  return phrases.map((p, i) => {
    54→    const words = p.trim().split(/\s+/);
    55→    if (words.length <= 3) return '...';
    56→    const mode = i % 3; // 0=start, 1=middle, 2=end
    57→    if (mode === 0) {
    58→      const count = Math.max(2, Math.ceil(words.length * 0.35));
    59→      return words.slice(0, count).join(' ') + ' ...';
    60→    } else if (mode === 1) {
    61→      const startIdx = Math.floor(words.length * 0.3);
    62→      const count = Math.max(2, Math.ceil(words.length * 0.4));
    63→      return '... ' + words.slice(startIdx, startIdx + count).join(' ') + ' ...';
    64→    } else {
    65→      const startIdx = Math.max(0, words.length - Math.ceil(words.length * 0.4));
    66→      return '... ' + words.slice(startIdx).join(' ');
    67→    }
    68→  });
    69→}

    71→// Minimal typings for the Web Speech API (Google's free STT in Chrome)
    72→interface SpeechRecognitionResultLike {
    73→  0: { transcript: string };
    74→  isFinal: boolean;
    75→}
    76→interface SpeechRecognitionEventLike {
    77→  results: { length: number; [index: number]: SpeechRecognitionResultLike };
    78→  resultIndex: number;
    79→}
    80→interface SpeechRecognitionLike {
    81→  lang: string;
    82→  continuous: boolean;
    83→  interimResults: boolean;
    84→  maxAlternatives: number;
    85→  start: () => void;
    86→  stop: () => void;
    87→  abort: () => void;
    88→  onresult: ((e: SpeechRecognitionEventLike) => void) | null;
    89→  onerror: ((e: any) => void) | null;
    90→  onend: (() => void) | null;
    91→  onstart: (() => void) | null;
    92→}
    93→type SpeechRecognitionCtor = new () => SpeechRecognitionLike;

    95→export function ConversationChat({ topic, onExit }: ConversationChatProps) {

   100→  const isVorstellung = topic.id === 'b1-vorstellung';

   107→        // Show first 5-7 words of each chunk as a hint

   111→      const vocabulary: { de: string; tr: string }[] = [

   113→        { de: 'Ich komme aus...', tr: 'Ben ...den geliyorum' },
   114→        { de: 'Ich wohne in...', tr: 'Ben ...da yaşıyorum' },
   115→        { de: 'Ich bin ... Jahre alt', tr: 'Ben ... yaşındayım' },
   116→        { de: 'Ich arbeite als...', tr: 'Ben ... olarak çalışıyorum' },
   117→        { de: 'In meiner Freizeit...', tr: 'Boş zamanlarımda...' },
   118→        { de: 'Ich lerne Deutsch, weil...', tr: 'Almanca öğreniyorum çünkü...' },
   119→        { de: 'Mein Hobby ist...', tr: 'Hobim ...' },
   120→      ];
   121→      return { ...topic, systemPrompt, opener, samplePhrases, vocabulary };
   122→    }
   123→    return topic;
   124→  })();
   125→
   126→  // Hint difficulty selection
   127→  const [hintDifficulty, setHintDifficulty] = useState<HintDifficulty | null>(null);
   128→  const hintDifficultyRef = useRef<HintDifficulty | null>(null);
   129→  // Keep ref in sync with state (avoids stale closures in callbacks)
   130→  useEffect(() => { hintDifficultyRef.current = hintDifficulty; }, [hintDifficulty]);
   131→
   132→  const [messages, setMessages] = useState<ChatMessage[]>([]);
   133→  const [phase, setPhase] = useState<Phase>('chatting');
   134→  const [textInput, setTextInput] = useState('');
   135→  const [showHints, setShowHints] = useState(false);
   136→  const [showVocab, setShowVocab] = useState(false);
   137→  const [error, setError] = useState<string | null>(null);
   138→  const [liveTranscript, setLiveTranscript] = useState<string>('');
   139→  const [browserSupported, setBrowserSupported] = useState<boolean>(true);
   140→  const [showTypingHint, setShowTypingHint] = useState(false);
   141→  const typingHintTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   142→
   143→  // ===== Google Web Speech API quota tracking =====
   144→  const speechUsage = useAppStore((s) => s.speechUsage);
   145→  const incrementSpeechUsage = useAppStore((s) => s.incrementSpeechUsage);
   146→  const hasSpeechQuota = useAppStore((s) => s.hasSpeechQuota);
   147→  const remainingSpeechQuota = useAppStore((s) => s.remainingSpeechQuota);
   148→  const daysUntilSpeechReset = useAppStore((s) => s.daysUntilSpeechReset);
   149→
   150→  // Recompute remaining on every render — store getters read fresh state
   151→  const quotaRemaining = remainingSpeechQuota();
   152→  const quotaExhausted = !hasSpeechQuota();

   154→
   155→  const tts = useTTS({ lang: 'de-DE' });
   156→  const messagesEndRef = useRef<HTMLDivElement | null>(null);
   157→  const startTimeRef = useRef<number>(Date.now());
   158→  const autoPlayRef = useRef<boolean>(true);
   159→  const lastAiMessageRef = useRef<ChatMessage | null>(null);
   160→
   161→  // SpeechRecognition refs
   162→  const recognitionRef = useRef<SpeechRecognitionLike | null>(null);
   163→  const finalTranscriptRef = useRef<string>('');
   164→  const manualStopRef = useRef<boolean>(false);
   165→  const maxDurationTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   166→
   167→  // Lazy-init the SpeechRecognition constructor (browser only)
   168→  const getRecognitionCtor = useCallback((): SpeechRecognitionCtor | null => {
   169→    if (typeof window === 'undefined') return null;
   170→    const w = window as any;
   171→    return ((w.SpeechRecognition || w.webkitSpeechRecognition) as
   172→      | SpeechRecognitionCtor
   173→      | undefined) ?? null;
   174→  }, []);
   175→
   176→  // Typing hint: show after 3s of inactivity in 'chatting' phase
   177→  useEffect(() => {
   178→    // Clear any existing timer
   179→    if (typingHintTimerRef.current) {
   180→      clearTimeout(typingHintTimerRef.current);
   181→      typingHintTimerRef.current = null;
   182→    }
   183→    setShowTypingHint(false);
   184→
   185→    if (phase === 'chatting' && messages.length > 0) {
   186→      typingHintTimerRef.current = setTimeout(() => {
   187→        setShowTypingHint(true);
   188→      }, 3000);
   189→    }
   190→
   191→    return () => {
   192→      if (typingHintTimerRef.current) {
   193→        clearTimeout(typingHintTimerRef.current);
   194→        typingHintTimerRef.current = null;
   195→      }
   196→    };
   197→  }, [phase, messages.length]);
   198→
   199→  // Auto-show hints when AI returns NEW suggestions
   200→  // In basic mode: always auto-show (hints are meant to be always visible)
   201→  // In middle/high mode: respect userClosedHintsRef (don't reopen if user closed them)
   202→  const userClosedHintsRef = useRef(false);
   203→  const lastAutoShowMsgIdRef = useRef<string>('');
   204→  useEffect(() => {
   205→    if (phase !== 'chatting') return;
   206→    const lastMsg = messages[messages.length - 1];
   207→    if (!lastMsg || lastMsg.sender !== 'ai') return;
   208→    if (!lastMsg.suggestions || lastMsg.suggestions.length === 0) return;
   209→    if (messages.length <= 1) return; // skip opener
   210→    // Only auto-show for NEW AI messages (not re-triggers on same message)
   211→    if (lastAutoShowMsgIdRef.current === lastMsg.id) return;
   212→    lastAutoShowMsgIdRef.current = lastMsg.id;
   213→    // In basic mode, always show. In other modes, respect user preference.
   214→    if (hintDifficultyRef.current === 'basic' || !userClosedHintsRef.current) {
   215→      setShowHints(true);
   216→    }
   217→  }, [messages, phase]);
   218→
   219→  // Check support on mount
   220→  useEffect(() => {
   221→    const ctor = getRecognitionCtor();
   222→    if (!ctor) {
   223→      setBrowserSupported(false);
   224→    }
   225→  }, [getRecognitionCtor]);
   226→
   227→  // Auto-scroll to bottom on new message
   228→  useEffect(() => {
   229→    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth', block: 'end' });
   230→  }, [messages, phase, liveTranscript]);
   231→
   232→  // Auto-play AI messages via browser TTS (Google's free voices in Chrome)
   233→  useEffect(() => {
   234→    if (!autoPlayRef.current) return;
   235→    const lastMsg = messages[messages.length - 1];
   236→    if (!lastMsg || lastMsg.sender !== 'ai') return;
   237→    if (lastMsg === lastAiMessageRef.current) return;
   238→    lastAiMessageRef.current = lastMsg;
   239→
   240→    // Browser-only TTS (SpeechSynthesis — Google's free Web Speech API)
   241→    tts.speak(lastMsg.text).catch((e) =>
   242→      console.warn('Browser TTS failed:', e)

   244→  }, [messages, tts]);
   245→
   246→  const fetchAiReply = useCallback(async (history: ChatMessage[]) => {
   247→    setPhase('loading-ai');
   248→    setError(null);
   249→    try {
   250→      const apiMessages = history
   251→        .filter((m) => m.id !== 'opener')
   252→        .map((m) => ({
   253→          role: m.sender === 'user' ? 'user' : 'assistant',
   254→          content: m.text,
   255→        }));
   256→      apiMessages.unshift({ role: 'assistant', content: effectiveTopic.opener });
   257→
   258→      const userTurnIndex = history.filter((m) => m.sender === 'user').length;
   259→
   260→      const { zhipuKey, openaiKey, claudeKey, googleAiKey } = useAppStore.getState().apiKeys;
   261→
   262→      const response = await fetch('/api/chat', {
   263→        method: 'POST',
   264→        headers: { 'Content-Type': 'application/json' },
   265→        body: JSON.stringify({
   266→          systemPrompt:
   267→            effectiveTopic.systemPrompt +
   268→            `\n\nBu konuşma ${MAX_TURNS} turdan oluşacak. ${
   269→              userTurnIndex >= MAX_TURNS - 1
   270→                ? 'Bu son tur — vedalaşarak bitir.'
   271→                : ''
   272→            }`,
   273→          messages: apiMessages,
   274→          temperature: 0.7,
   275→          maxTokens: isVorstellung ? 800 : 600,
   276→          topicId: topic.id,
   277→          zhipuKey: zhipuKey || undefined,
   278→          openaiKey: openaiKey || undefined,
   279→          claudeKey: claudeKey || undefined,
   280→          googleAiKey: googleAiKey || undefined,
   281→        }),
   282→      });
   283→      if (!response.ok) {
   284→        const errData = await response.json().catch(() => ({}));
   285→        throw new Error(errData.error || errData.detail || `HTTP ${response.status}`);
   286→      }
   287→      const data = await response.json() as { reply?: string; tip?: string; isEnding?: boolean; suggestions?: string[] };
   288→      const aiMsg: ChatMessage = {
   289→        id: `ai-${Date.now()}`,
   290→        sender: 'ai',
   291→        text: (data.reply || '').trim() || '...',
   292→        tip: data.tip || '',
   293→        isEnding: Boolean(data.isEnding) || userTurnIndex >= MAX_TURNS - 1,
   294→        suggestions: Array.isArray(data.suggestions) && data.suggestions.length > 0 ? data.suggestions : undefined,
   295→        timestamp: Date.now(),
   296→      };
   297→      setMessages((prev) => [...prev, aiMsg]);
   298→
   299→      if (aiMsg.isEnding) {
   300→        setTimeout(() => setPhase('ended'), 1500);
   301→      } else {
   302→        setPhase('chatting');
   303→      }
   304→    } catch (err) {
   305→      console.error('[ConversationChat] AI reply failed:', err);
   306→      setError('Yanıt alınamadı: ' + (err as Error).message);
   307→      setPhase('chatting');
   308→    }
   309→  }, [effectiveTopic]);
   310→
   311→  // ===== Browser SpeechRecognition (Google's free STT) =====
   312→  const startListening = useCallback(async () => {
   313→    const ctor = getRecognitionCtor();
   314→    if (!ctor) {
   315→      setBrowserSupported(false);
   316→      setError('Tarayıcı ses tanımayı desteklemiyor. Chrome veya Edge kullanın. Yazarak cevap verebilirsiniz.');
   317→      return;
   318→    }
   319→
   320→    // Quota check — if exhausted, refuse to start and tell user when it resets
   321→    if (!hasSpeechQuota()) {
   322→      setError(
   323→        `Bu ayki ücretsiz Google Web Speech API kotan doldu (${speechUsage.monthlyLimit} tanıma). ` +
   324→        `Kota her ayın 1'inde otomatik yenilenir. ${daysUntilReset} gün sonra tekrar kullanabilirsin. ` +
   325→        `Bu süre içinde yazarak cevap vermeye devam edebilirsin.`
   326→      );
   327→      return;
   328→    }
   329→
   330→    // Stop any playing audio before recording
   331→    tts.stop();
   332→    setError(null);
   333→    // Close hints in middle/high mode when user starts speaking (basic keeps them visible)
   334→    const diff = hintDifficultyRef.current;
   335→    if (diff !== 'basic') {
   336→      setShowHints(false);
   337→      userClosedHintsRef.current = true;
   338→    }
   339→    setLiveTranscript('');
   340→    finalTranscriptRef.current = '';
   341→    manualStopRef.current = false;
   342→
   343→    // Stop any existing recognition
   344→    if (recognitionRef.current) {
   345→      try { recognitionRef.current.abort(); } catch {}
   346→    }
   347→
   348→    const recognition = new ctor();
   349→    recognition.lang = 'de-DE';
   350→    recognition.continuous = true;
   351→    recognition.interimResults = true;
   352→    recognition.maxAlternatives = 1;
   353→
   354→    recognition.onstart = () => {

   356→    };
   357→
   358→    recognition.onresult = (event: SpeechRecognitionEventLike) => {
   359→      let interim = '';
   360→      let final = finalTranscriptRef.current;
   361→      for (let i = event.resultIndex; i < event.results.length; i++) {
   362→        const result = event.results[i];
   363→        const txt = result[0].transcript;
   364→        if (result.isFinal) {
   365→          final += txt + ' ';
   366→        } else {
   367→          interim += txt;
   368→        }
   369→      }
   370→      finalTranscriptRef.current = final;
   371→      setLiveTranscript((final + interim).trim());
   372→    };
   373→
   374→    recognition.onerror = (e: any) => {
   375→      console.error('[ConversationChat] SpeechRecognition error:', e);
   376→      if (e.error === 'not-allowed' || e.error === 'service-not-allowed') {
   377→        setError('Mikrofon izni reddedildi. Tarayıcı ayarlarından izin verin.');
   378→      } else if (e.error === 'no-speech') {
   379→        if (!finalTranscriptRef.current.trim()) {
   380→          setError('Ses algılanamadı. Tekrar deneyin veya yazarak cevap verin.');
   381→        }
   382→      } else if (e.error === 'network') {
   383→        setError('Ağ hatası. İnternet bağlantınızı kontrol edin (Web Speech API internet gerektirir).');
   384→      } else if (e.error === 'aborted') {
   385→        // Manual abort — not an error
   386→        return;
   387→      } else {
   388→        setError(`Tanıma hatası: ${e.error}`);
   389→      }
   390→      setPhase('chatting');
   391→    };
   392→
   393→    recognition.onend = () => {
   394→      if (maxDurationTimerRef.current) {
   395→        clearTimeout(maxDurationTimerRef.current);
   396→        maxDurationTimerRef.current = null;
   397→      }
   398→      const finalText = finalTranscriptRef.current.trim();
   399→      if (finalText) {
   400→        // Successful STT call — increment monthly counter
   401→        incrementSpeechUsage();
   402→
   403→        // Close hints immediately when speech ends (before AI reply)
   404→        setShowHints(false);
   405→        userClosedHintsRef.current = true;
   406→
   407→        // Add it as user message and fetch AI reply
   408→        const userMsg: ChatMessage = {
   409→          id: `user-${Date.now()}`,
   410→          sender: 'user',
   411→          text: finalText,
   412→          timestamp: Date.now(),
   413→        };
   414→        setMessages((prev) => [...prev, userMsg]);
   415→        setLiveTranscript('');
   416→        // Trigger AI reply with the updated history
   417→        fetchAiReply([...messages, userMsg]);
   418→      } else if (manualStopRef.current) {
   419→        // User stopped but no transcript — don't count this against quota
   420→        setPhase('chatting');
   421→      }
   422→    };
   423→
   424→    recognitionRef.current = recognition;
   425→
   426→    try {
   427→      recognition.start();
   428→      // Auto-stop after 30 seconds to avoid runaway listening
   429→      maxDurationTimerRef.current = setTimeout(() => {
   430→        manualStopRef.current = true;
   431→        try { recognition.stop(); } catch {}
   432→      }, 30_000);
   433→    } catch (err) {
   434→      console.error('[ConversationChat] start failed:', err);
   435→      setError('Dinleme başlatılamadı. Yazarak cevap verebilirsiniz.');
   436→      setPhase('chatting');
   437→    }
   438→  }, [getRecognitionCtor, messages, tts, fetchAiReply, hasSpeechQuota, speechUsage.monthlyLimit, daysUntilReset, incrementSpeechUsage]);
   439→
   440→  const stopListening = useCallback(() => {
   441→    manualStopRef.current = true;
   442→    if (maxDurationTimerRef.current) {
   443→      clearTimeout(maxDurationTimerRef.current);
   444→      maxDurationTimerRef.current = null;
   445→    }
   446→    if (recognitionRef.current) {
   447→      try { recognitionRef.current.stop(); } catch {}
   448→    }
   449→  }, []);
   450→
   451→  // Mic toggle
   452→  const handleMicToggle = useCallback(async () => {
   453→    if (phase === 'listening') {
   454→      stopListening();
   455→    } else if (phase === 'chatting') {
   456→      await startListening();
   457→    }
   458→  }, [phase, startListening, stopListening]);
   459→
   460→  const handleSendText = useCallback(async () => {
   461→    const text = textInput.trim();
   462→    if (!text || phase !== 'chatting') return;
   463→    setTextInput('');
   464→    // Close hints in middle/high mode (basic keeps them visible)
   465→    const diff = hintDifficultyRef.current;
   466→    if (diff !== 'basic') {
   467→      setShowHints(false);
   468→      userClosedHintsRef.current = true;
   469→    }
   470→    const userMsg: ChatMessage = {
   471→      id: `user-${Date.now()}`,
   472→      sender: 'user',
   473→      text,
   474→      timestamp: Date.now(),
   475→    };
   476→    setMessages((prev) => [...prev, userMsg]);
   477→    await fetchAiReply([...messages, userMsg]);
   478→  }, [textInput, phase, messages, fetchAiReply]);
   479→
   480→  const handleEndConversation = useCallback(() => {
   481→    // Stop any active recognition
   482→    if (recognitionRef.current) {
   483→      try { recognitionRef.current.abort(); } catch {}
   484→    }
   485→    tts.stop();
   486→    setPhase('ended');
   487→  }, [tts]);
   488→
   489→  // Build and send the summary when phase becomes 'ended'
   490→  useEffect(() => {
   491→    if (phase !== 'ended') return;
   492→    const userMessageCount = messages.filter((m) => m.sender === 'user').length;
   493→    const tipsCount = messages.filter((m) => m.tip && m.tip.trim()).length;
   494→    const summary: ConversationSummary = {
   495→      topicId: topic.id,
   496→      topicTitleDe: topic.titleDe,
   497→      topicTitleTr: topic.titleTr,
   498→      messageCount: messages.length,
   499→      userMessageCount,
   500→      tipsCount,
   501→      durationSec: Math.round((Date.now() - startTimeRef.current) / 1000),
   502→      hintDifficulty: hintDifficulty || 'unknown',
   503→      messages,
   504→    };
   505→    onExit(summary);
   506→  }, [phase, messages, topic, onExit]);
   507→
   508→  // Replay an AI message via browser TTS
   509→  const replayMessage = useCallback((msg: ChatMessage) => {
   510→    tts.speak(msg.text);
   511→  }, [tts]);
   512→
   513→  // Get dynamic suggestions from the LAST AI message only (not older ones — prevents stale context)
   514→  const lastMsg = messages[messages.length - 1];
   515→  const lastAiSuggestions = (lastMsg?.sender === 'ai' && lastMsg.suggestions?.length) ? lastMsg.suggestions : undefined;
   516→
   517→  // Apply hint difficulty filtering
   518→  // Priority: AI suggestions (dynamic) > static samplePhrases (fallback)
   519→  const rawHints = lastAiSuggestions || (effectiveTopic.samplePhrases.length > 0 ? effectiveTopic.samplePhrases : undefined);
   520→  const inlineHints = (() => {
   521→    if (!rawHints || !hintDifficulty) return rawHints;
   522→    if (hintDifficulty === 'high') return makeStablePartialHints(rawHints);
