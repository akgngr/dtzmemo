     1→     1→'use client';

     4→     4→import { motion, AnimatePresence } from 'framer-motion';
     5→     5→import {

     8→     8→} from 'lucide-react';

    10→    10→import { Button } from '@/components/ui/button';
    11→    11→import { Badge } from '@/components/ui/badge';
    12→    12→import { type ConversationTopic } from '@/lib/conversation-topics';
    13→    13→import { useTTS } from '@/hooks/use-tts';
    14→    14→import { useAppStore } from '@/lib/store';
    15→    15→import { cn } from '@/lib/utils';
    16→    16→
    17→    17→type Sender = 'user' | 'ai';
    18→    18→
    19→    19→interface ChatMessage {
    20→    20→  id: string;
    21→    21→  sender: Sender;
    22→    22→  text: string;
    23→    23→  tip?: string;
    24→    24→  isEnding?: boolean;
    25→    25→  timestamp: number;
    26→    26→}
    27→    27→
    28→    28→interface ConversationChatProps {
    29→    29→  topic: ConversationTopic;
    30→    30→  onExit: (summary: ConversationSummary) => void;
    31→    31→}
    32→    32→
    33→    33→export interface ConversationSummary {
    34→    34→  topicId: string;
    35→    35→  topicTitleDe: string;
    36→    36→  topicTitleTr: string;
    37→    37→  messageCount: number;
    38→    38→  userMessageCount: number;
    39→    39→  tipsCount: number;
    40→    40→  durationSec: number;
    41→    41→  messages: ChatMessage[];
    42→    42→}
    43→    43→
    44→    44→type Phase = 'chatting' | 'listening' | 'loading-ai' | 'ended';
    45→    45→
    46→    46→const MAX_TURNS = 10;
    47→    47→
    48→    48→// Minimal typings for the Web Speech API (Google's free STT in Chrome)
    49→    49→interface SpeechRecognitionResultLike {
    50→    50→  0: { transcript: string };
    51→    51→  isFinal: boolean;
    52→    52→}
    53→    53→interface SpeechRecognitionEventLike {
    54→    54→  results: { length: number; [index: number]: SpeechRecognitionResultLike };
    55→    55→  resultIndex: number;
    56→    56→}
    57→    57→interface SpeechRecognitionLike {
    58→    58→  lang: string;
    59→    59→  continuous: boolean;
    60→    60→  interimResults: boolean;
    61→    61→  maxAlternatives: number;
    62→    62→  start: () => void;
    63→    63→  stop: () => void;
    64→    64→  abort: () => void;
    65→    65→  onresult: ((e: SpeechRecognitionEventLike) => void) | null;
    66→    66→  onerror: ((e: any) => void) | null;
    67→    67→  onend: (() => void) | null;
    68→    68→  onstart: (() => void) | null;
    69→    69→}
    70→    70→type SpeechRecognitionCtor = new () => SpeechRecognitionLike;
    71→    71→
    72→    72→export function ConversationChat({ topic, onExit }: ConversationChatProps) {
    73→    73→  const [messages, setMessages] = useState<ChatMessage[]>([
    74→    74→    {
    75→    75→      id: 'opener',
    76→    76→      sender: 'ai',
    77→    77→      text: topic.opener,
    78→    78→      timestamp: Date.now(),
    79→    79→    },
    80→    80→  ]);
    81→    81→  const [phase, setPhase] = useState<Phase>('chatting');
    82→    82→  const [textInput, setTextInput] = useState('');
    83→    83→  const [showHints, setShowHints] = useState(false);
    84→    84→  const [showVocab, setShowVocab] = useState(false);
    85→    85→  const [error, setError] = useState<string | null>(null);
    86→    86→  const [liveTranscript, setLiveTranscript] = useState<string>('');
    87→    87→  const [browserSupported, setBrowserSupported] = useState<boolean>(true);
    88→    88→
    89→    89→  // ===== Google Web Speech API quota tracking =====
    90→    90→  const speechUsage = useAppStore((s) => s.speechUsage);
    91→    91→  const incrementSpeechUsage = useAppStore((s) => s.incrementSpeechUsage);
    92→    92→  const hasSpeechQuota = useAppStore((s) => s.hasSpeechQuota);
    93→    93→  const remainingSpeechQuota = useAppStore((s) => s.remainingSpeechQuota);
    94→    94→  const daysUntilSpeechReset = useAppStore((s) => s.daysUntilSpeechReset);
    95→    95→
    96→    96→  // Recompute remaining on every render — store getters read fresh state
    97→    97→  const quotaRemaining = remainingSpeechQuota();
    98→    98→  const quotaExhausted = !hasSpeechQuota();
    99→    99→  const daysUntilReset = daysUntilSpeechReset();
   100→   100→
   101→   101→  const tts = useTTS({ lang: 'de-DE' });
   102→   102→  const messagesEndRef = useRef<HTMLDivElement | null>(null);
   103→   103→  const startTimeRef = useRef<number>(Date.now());
   104→   104→  const autoPlayRef = useRef<boolean>(true);
   105→   105→  const lastAiMessageRef = useRef<ChatMessage | null>(null);
   106→   106→
   107→   107→  // SpeechRecognition refs
   108→   108→  const recognitionRef = useRef<SpeechRecognitionLike | null>(null);
   109→   109→  const finalTranscriptRef = useRef<string>('');
   110→   110→  const manualStopRef = useRef<boolean>(false);
   111→   111→  const maxDurationTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   112→   112→
   113→   113→  // Lazy-init the SpeechRecognition constructor (browser only)
   114→   114→  const getRecognitionCtor = useCallback((): SpeechRecognitionCtor | null => {
   115→   115→    if (typeof window === 'undefined') return null;
   116→   116→    const w = window as any;
   117→   117→    return ((w.SpeechRecognition || w.webkitSpeechRecognition) as
   118→   118→      | SpeechRecognitionCtor
   119→   119→      | undefined) ?? null;
   120→   120→  }, []);
   121→   121→
   122→   122→  // Check support on mount
   123→   123→  useEffect(() => {
   124→   124→    const ctor = getRecognitionCtor();
   125→   125→    if (!ctor) {
   126→   126→      setBrowserSupported(false);
   127→   127→    }
   128→   128→  }, [getRecognitionCtor]);
   129→   129→
   130→   130→  // Auto-scroll to bottom on new message
   131→   131→  useEffect(() => {
   132→   132→    messagesEndRef.current?.scrollIntoView({ behavior: 'smooth', block: 'end' });
   133→   133→  }, [messages, phase, liveTranscript]);
   134→   134→
   135→   135→  // Auto-play AI messages via browser TTS (Google's free voices in Chrome)
   136→   136→  useEffect(() => {
   137→   137→    if (!autoPlayRef.current) return;
   138→   138→    const lastMsg = messages[messages.length - 1];
   139→   139→    if (!lastMsg || lastMsg.sender !== 'ai') return;
   140→   140→    if (lastMsg === lastAiMessageRef.current) return;
   141→   141→    lastAiMessageRef.current = lastMsg;
   142→   142→
   143→   143→    // Browser-only TTS (SpeechSynthesis — Google's free Web Speech API)
   144→   144→    tts.speak(lastMsg.text).catch((e) =>
   145→   145→      console.warn('Browser TTS failed:', e)
   146→   146→    );
   147→   147→  }, [messages, tts]);
   148→   148→
   149→   149→  const fetchAiReply = useCallback(async (history: ChatMessage[]) => {
   150→   150→    setPhase('loading-ai');
   151→   151→    setError(null);
   152→   152→    try {
   153→   153→      const apiMessages = history
   154→   154→        .filter((m) => m.id !== 'opener')
   155→   155→        .map((m) => ({
   156→   156→          role: m.sender === 'user' ? 'user' : 'assistant',
   157→   157→          content: m.text,
   158→   158→        }));
   159→   159→      apiMessages.unshift({ role: 'assistant', content: topic.opener });
   160→   160→
   161→   161→      const userTurnIndex = history.filter((m) => m.sender === 'user').length;
   162→   162→
   163→   163→      const { zhipuKey, openaiKey, claudeKey, googleAiKey } = useAppStore.getState().apiKeys;
   164→   164→
   165→   165→      const response = await fetch('/api/chat', {
   166→   166→        method: 'POST',
   167→   167→        headers: { 'Content-Type': 'application/json' },
   168→   168→        body: JSON.stringify({
   169→   169→          systemPrompt:
   170→   170→            topic.systemPrompt +
   171→   171→            `\n\nBu konuşma ${MAX_TURNS} turdan oluşacak. ${
   172→   172→              userTurnIndex >= MAX_TURNS - 1
   173→   173→                ? 'Bu son tur — vedalaşarak bitir.'
   174→   174→                : ''
   175→   175→            }`,
   176→   176→          messages: apiMessages,
   177→   177→          temperature: 0.7,
   178→   178→          maxTokens: 400,
   179→   179→          zhipuKey: zhipuKey || undefined,
   180→   180→          openaiKey: openaiKey || undefined,
   181→   181→          claudeKey: claudeKey || undefined,
   182→   182→          googleAiKey: googleAiKey || undefined,
   183→   183→        }),
   184→   184→      });
   185→   185→      if (!response.ok) {
   186→   186→        const errData = await response.json().catch(() => ({}));
   187→   187→        throw new Error(errData.error || errData.detail || `HTTP ${response.status}`);
   188→   188→      }
   189→   189→      const data = await response.json();
   190→   190→      const aiMsg: ChatMessage = {
   191→   191→        id: `ai-${Date.now()}`,
   192→   192→        sender: 'ai',
   193→   193→        text: (data.reply || '').trim() || '...',
   194→   194→        tip: data.tip || '',
   195→   195→        isEnding: Boolean(data.isEnding) || userTurnIndex >= MAX_TURNS - 1,
   196→   196→        timestamp: Date.now(),
   197→   197→      };
   198→   198→      setMessages((prev) => [...prev, aiMsg]);
   199→   199→
   200→   200→      if (aiMsg.isEnding) {
   201→   201→        setTimeout(() => setPhase('ended'), 1500);
   202→   202→      } else {
   203→   203→        setPhase('chatting');
   204→   204→      }
   205→   205→    } catch (err) {
   206→   206→      console.error('[ConversationChat] AI reply failed:', err);
   207→   207→      setError('Yanıt alınamadı: ' + (err as Error).message);
   208→   208→      setPhase('chatting');
   209→   209→    }
   210→   210→  }, [topic]);
   211→   211→
   212→   212→  // ===== Browser SpeechRecognition (Google's free STT) =====
   213→   213→  const startListening = useCallback(async () => {
   214→   214→    const ctor = getRecognitionCtor();
   215→   215→    if (!ctor) {
   216→   216→      setBrowserSupported(false);
   217→   217→      setError('Tarayıcı ses tanımayı desteklemiyor. Chrome veya Edge kullanın. Yazarak cevap verebilirsiniz.');
   218→   218→      return;
   219→   219→    }
   220→   220→
   221→   221→    // Quota check — if exhausted, refuse to start and tell user when it resets
   222→   222→    if (!hasSpeechQuota()) {
   223→   223→      setError(
   224→   224→        `Bu ayki ücretsiz Google Web Speech API kotan doldu (${speechUsage.monthlyLimit} tanıma). ` +
   225→   225→        `Kota her ayın 1'inde otomatik yenilenir. ${daysUntilReset} gün sonra tekrar kullanabilirsin. ` +
   226→   226→        `Bu süre içinde yazarak cevap vermeye devam edebilirsin.`
   227→   227→      );
   228→   228→      return;
   229→   229→    }
   230→   230→
   231→   231→    // Stop any playing audio before recording
   232→   232→    tts.stop();
   233→   233→    setError(null);
   234→   234→    setLiveTranscript('');
   235→   235→    finalTranscriptRef.current = '';
   236→   236→    manualStopRef.current = false;
   237→   237→
   238→   238→    // Stop any existing recognition
   239→   239→    if (recognitionRef.current) {
   240→   240→      try { recognitionRef.current.abort(); } catch {}
   241→   241→    }
   242→   242→
   243→   243→    const recognition = new ctor();
   244→   244→    recognition.lang = 'de-DE';
   245→   245→    recognition.continuous = true;
   246→   246→    recognition.interimResults = true;
   247→   247→    recognition.maxAlternatives = 1;
   248→   248→
   249→   249→    recognition.onstart = () => {
   250→   250→      setPhase('listening');
   251→   251→    };
   252→   252→
   253→   253→    recognition.onresult = (event: SpeechRecognitionEventLike) => {
   254→   254→      let interim = '';
   255→   255→      let final = finalTranscriptRef.current;
   256→   256→      for (let i = event.resultIndex; i < event.results.length; i++) {
   257→   257→        const result = event.results[i];
   258→   258→        const txt = result[0].transcript;
   259→   259→        if (result.isFinal) {
   260→   260→          final += txt + ' ';
   261→   261→        } else {
   262→   262→          interim += txt;
   263→   263→        }
   264→   264→      }
   265→   265→      finalTranscriptRef.current = final;
   266→   266→      setLiveTranscript((final + interim).trim());
   267→   267→    };
   268→   268→
   269→   269→    recognition.onerror = (e: any) => {
   270→   270→      console.error('[ConversationChat] SpeechRecognition error:', e);
   271→   271→      if (e.error === 'not-allowed' || e.error === 'service-not-allowed') {
   272→   272→        setError('Mikrofon izni reddedildi. Tarayıcı ayarlarından izin verin.');
   273→   273→      } else if (e.error === 'no-speech') {
   274→   274→        if (!finalTranscriptRef.current.trim()) {
   275→   275→          setError('Ses algılanamadı. Tekrar deneyin veya yazarak cevap verin.');
   276→   276→        }
   277→   277→      } else if (e.error === 'network') {
   278→   278→        setError('Ağ hatası. İnternet bağlantınızı kontrol edin (Web Speech API internet gerektirir).');
   279→   279→      } else if (e.error === 'aborted') {
   280→   280→        // Manual abort — not an error
   281→   281→        return;
   282→   282→      } else {
   283→   283→        setError(`Tanıma hatası: ${e.error}`);
   284→   284→      }
   285→   285→      setPhase('chatting');
   286→   286→    };
   287→   287→
   288→   288→    recognition.onend = () => {
   289→   289→      if (maxDurationTimerRef.current) {
   290→   290→        clearTimeout(maxDurationTimerRef.current);
   291→   291→        maxDurationTimerRef.current = null;
   292→   292→      }
   293→   293→      const finalText = finalTranscriptRef.current.trim();
   294→   294→      if (finalText) {
   295→   295→        // Successful STT call — increment monthly counter
   296→   296→        incrementSpeechUsage();
   297→   297→
   298→   298→        // Add it as user message and fetch AI reply
   299→   299→        const userMsg: ChatMessage = {
   300→   300→          id: `user-${Date.now()}`,
   301→   301→          sender: 'user',
   302→   302→          text: finalText,
   303→   303→          timestamp: Date.now(),
   304→   304→        };
   305→   305→        setMessages((prev) => [...prev, userMsg]);
   306→   306→        setLiveTranscript('');
   307→   307→        // Trigger AI reply with the updated history
   308→   308→        fetchAiReply([...messages, userMsg]);
   309→   309→      } else if (manualStopRef.current) {
   310→   310→        // User stopped but no transcript — don't count this against quota
   311→   311→        setPhase('chatting');
   312→   312→      }
   313→   313→    };
   314→   314→
   315→   315→    recognitionRef.current = recognition;
   316→   316→
   317→   317→    try {
   318→   318→      recognition.start();
   319→   319→      // Auto-stop after 30 seconds to avoid runaway listening
   320→   320→      maxDurationTimerRef.current = setTimeout(() => {
   321→   321→        manualStopRef.current = true;
   322→   322→        try { recognition.stop(); } catch {}
   323→   323→      }, 30_000);
   324→   324→    } catch (err) {
   325→   325→      console.error('[ConversationChat] start failed:', err);
   326→   326→      setError('Dinleme başlatılamadı. Yazarak cevap verebilirsiniz.');
   327→   327→      setPhase('chatting');
   328→   328→    }
   329→   329→  }, [getRecognitionCtor, messages, tts, fetchAiReply, hasSpeechQuota, speechUsage.monthlyLimit, daysUntilReset, incrementSpeechUsage]);
   330→   330→
   331→   331→  const stopListening = useCallback(() => {
   332→   332→    manualStopRef.current = true;
   333→   333→    if (maxDurationTimerRef.current) {
   334→   334→      clearTimeout(maxDurationTimerRef.current);
   335→   335→      maxDurationTimerRef.current = null;
   336→   336→    }
   337→   337→    if (recognitionRef.current) {
   338→   338→      try { recognitionRef.current.stop(); } catch {}
   339→   339→    }
   340→   340→  }, []);
   341→   341→
   342→   342→  // Mic toggle
   343→   343→  const handleMicToggle = useCallback(async () => {
   344→   344→    if (phase === 'listening') {
   345→   345→      stopListening();
   346→   346→    } else if (phase === 'chatting') {
   347→   347→      await startListening();
   348→   348→    }
   349→   349→  }, [phase, startListening, stopListening]);
   350→   350→
   351→   351→  const handleSendText = useCallback(async () => {
   352→   352→    const text = textInput.trim();
   353→   353→    if (!text || phase !== 'chatting') return;
   354→   354→    setTextInput('');
   355→   355→    const userMsg: ChatMessage = {
   356→   356→      id: `user-${Date.now()}`,
   357→   357→      sender: 'user',
   358→   358→      text,
   359→   359→      timestamp: Date.now(),
   360→   360→    };
   361→   361→    setMessages((prev) => [...prev, userMsg]);
   362→   362→    await fetchAiReply([...messages, userMsg]);
   363→   363→  }, [textInput, phase, messages, fetchAiReply]);
   364→   364→
   365→   365→  const handleEndConversation = useCallback(() => {
   366→   366→    // Stop any active recognition
   367→   367→    if (recognitionRef.current) {
   368→   368→      try { recognitionRef.current.abort(); } catch {}
   369→   369→    }
   370→   370→    tts.stop();
   371→   371→    setPhase('ended');
   372→   372→  }, [tts]);
   373→   373→
   374→   374→  // Build and send the summary when phase becomes 'ended'
   375→   375→  useEffect(() => {
   376→   376→    if (phase !== 'ended') return;
   377→   377→    const userMessageCount = messages.filter((m) => m.sender === 'user').length;
   378→   378→    const tipsCount = messages.filter((m) => m.tip && m.tip.trim()).length;
   379→   379→    const summary: ConversationSummary = {
   380→   380→      topicId: topic.id,
   381→   381→      topicTitleDe: topic.titleDe,
   382→   382→      topicTitleTr: topic.titleTr,
   383→   383→      messageCount: messages.length,
   384→   384→      userMessageCount,
   385→   385→      tipsCount,
   386→   386→      durationSec: Math.round((Date.now() - startTimeRef.current) / 1000),
   387→   387→      messages,
   388→   388→    };
   389→   389→    onExit(summary);
   390→   390→  }, [phase, messages, topic, onExit]);
   391→   391→
   392→   392→  // Replay an AI message via browser TTS
   393→   393→  const replayMessage = useCallback((msg: ChatMessage) => {
   394→   394→    tts.speak(msg.text);
   395→   395→  }, [tts]);
   396→   396→
   397→   397→  const isBusy = phase === 'listening' || phase === 'loading-ai';
   398→   398→
   399→   399→  return (
   400→   400→    <div className="flex flex-col h-[calc(100vh-12rem)] md:h-[calc(100vh-8rem)]">
   401→   401→      {/* Topic header */}
   402→   402→      <div className={cn('rounded-xl p-3 text-white shadow-sm', topic.color)}>
   403→   403→        <div className="flex items-center justify-between gap-2">
   404→   404→          <div className="flex items-center gap-2 min-w-0">
   405→   405→            <span className="text-2xl">{topic.icon}</span>
   406→   406→            <div className="min-w-0">
   407→   407→              <h2 className="text-base font-bold leading-tight truncate">{topic.titleDe}</h2>
   408→   408→              <p className="text-xs opacity-90 truncate">{topic.titleTr} · {topic.difficulty}</p>
   409→   409→            </div>
   410→   410→          </div>
   411→   411→          <div className="flex items-center gap-1 shrink-0">
   412→   412→            <Button
   413→   413→              size="sm"
   414→   414→              variant="ghost"
   415→   415→              className="text-white hover:bg-white/20 h-8 px-2"
   416→   416→              onClick={() => setShowVocab((v) => !v)}
   417→   417→              aria-label="Kelime listesi"
   418→   418→            >
   419→   419→              <Lightbulb className="h-4 w-4" />
   420→   420→            </Button>
   421→   421→            <Button
   422→   422→              size="sm"
   423→   423→              variant="ghost"
   424→   424→              className="text-white hover:bg-white/20 h-8 px-2"
   425→   425→              onClick={() => setShowHints((v) => !v)}
   426→   426→              aria-label="Örnek cümleler"
   427→   427→            >
   428→   428→              <MessageCircle className="h-4 w-4" />
   429→   429→            </Button>
   430→   430→            <Button
   431→   431→              size="sm"
   432→   432→              variant="ghost"
   433→   433→              className="text-white hover:bg-white/20 h-8 px-2"
   434→   434→              onClick={handleEndConversation}
   435→   435→            >
   436→   436→              <X className="h-4 w-4" />
   437→   437→            </Button>
   438→   438→          </div>
   439→   439→        </div>
   440→   440→      </div>
   441→   441→
   442→   442→      {/* Vocabulary drawer (collapsible) */}
   443→   443→      <AnimatePresence>
   444→   444→        {showVocab && (
   445→   445→          <motion.div
   446→   446→            initial={{ opacity: 0, height: 0 }}
   447→   447→            animate={{ opacity: 1, height: 'auto' }}
   448→   448→            exit={{ opacity: 0, height: 0 }}
   449→   449→            className="overflow-hidden"
   450→   450→          >
   451→   451→            <Card className="border-0 shadow-sm mt-2">
   452→   452→              <CardContent className="p-3">
   453→   453→                <div className="flex items-center justify-between mb-2">
   454→   454→                  <span className="text-xs font-semibold uppercase text-muted-foreground">Önemli Kelimeler</span>
   455→   455→                  <button onClick={() => setShowVocab(false)} className="text-xs text-muted-foreground hover:text-foreground">
   456→   456→                    <X className="h-3 w-3" />
   457→   457→                  </button>
   458→   458→                </div>
   459→   459→                <div className="grid grid-cols-2 gap-1.5">
