    55→/** Simple heuristic: does a string contain Turkish-only special chars? */
    56→function hasTurkishSpecialChars(str: string): boolean {
    57→  return /[şığüşçöİŞĞÜŞÇÖ]/.test(str);
    58→}

    60→/** Check if a string looks like a CSV header (e.g. "german,turkish" or "Almanca,Türkçe") */
    61→function looksLikeHeader(first: string, second: string): boolean {
    62→  const headerKeywords = ['german', 'turkish', 'almanca', 'türkçe', 'deutsch', 'de', 'tr', 'kelime', 'word', 'çeviri', 'translation'];
    63→  const combined = (first + ' ' + second).toLowerCase();
    64→  return headerKeywords.some((k) => combined.includes(k));
    65→}

    67→/** Parse CSV content into word pairs */
    68→function parseCSV(content: string): CustomWord[] {
    69→  // Strip BOM if present
    70→  let cleaned = content.replace(/^\uFEFF/, '');
    71→
    72→  // Normalize line endings
    73→  cleaned = cleaned.replace(/\r\n/g, '\n').replace(/\r/g, '\n');
    74→
    75→  const lines = cleaned.split('\n').filter((line) => line.trim().length > 0);
    76→
    77→  if (lines.length === 0) return [];
    78→
    79→  let startIndex = 0;
    80→
    81→  // Check if the first line is a header
    82→  if (lines.length > 1) {
    83→    const firstLineParts = lines[0].split(',').map((s) => s.trim());
    84→    if (firstLineParts.length >= 2) {
    85→      const col1 = firstLineParts[0];
    86→      const col2 = firstLineParts[1];
    87→      // Skip if it looks like a header
    88→      if (looksLikeHeader(col1, col2) || hasTurkishSpecialChars(col1)) {
    89→        startIndex = 1;
    90→      }
    91→    }
    92→  }
    93→
    94→  const words: CustomWord[] = [];
    95→
    96→  for (let i = startIndex; i < lines.length; i++) {
    97→    const parts = lines[i].split(',').map((s) => s.trim());
    98→    if (parts.length >= 2 && parts[0] && parts[1]) {
    99→      words.push({ german: parts[0], turkish: parts[1] });
   100→    }
   101→  }
   102→
   103→  return words;
   104→}

   106→/** Encode a list to a base64 share string */
   107→function encodeList(list: Omit<CustomWordList, 'id' | 'createdAt' | 'category'>): string {
   108→  const payload = { n: list.name, d: list.description, w: list.words };
   109→  return btoa(JSON.stringify(payload));
   110→}
   111→
   112→/** Decode a base64 share string */
   113→function decodeList(encoded: string): { name: string; description: string; words: CustomWord[] } | null {
   114→  try {
   115→    const payload = JSON.parse(atob(encoded));
   116→    if (payload && payload.w && Array.isArray(payload.w)) {
   117→      return {
   118→        name: payload.n || 'İçe Aktarılan Liste',
   119→        description: payload.d || '',
   120→        words: payload.w.filter(
   121→          (w: any) => typeof w.german === 'string' && typeof w.turkish === 'string' && w.german && w.turkish
   122→        ),
   123→      };
   124→    }
   125→    return null;
   126→  } catch {
   127→    return null;
   128→  }
   129→}

   131→function generateShareURL(list: Omit<CustomWordList, 'id' | 'createdAt' | 'category'>): string {
   132→  const base64 = encodeList(list);
   133→  return window.location.origin + window.location.pathname + '?import=' + encodeURIComponent(base64);
   134→}

   136→async function copyToClipboard(text: string): Promise<boolean> {
   137→  try {
   138→    await navigator.clipboard.writeText(text);
   139→    return true;
   140→  } catch {
   141→    // Fallback for older browsers / restricted contexts
   142→    try {
   143→      const textarea = document.createElement('textarea');
   144→      textarea.value = text;
   145→      textarea.style.position = 'fixed';
   146→      textarea.style.opacity = '0';
   147→      document.body.appendChild(textarea);
   148→      textarea.select();
   149→      document.execCommand('copy');
   150→      document.body.removeChild(textarea);
   151→      return true;
   152→    } catch {
   153→      return false;
   154→    }
   155→  }
   156→}

   162→/** Build CSV string from word pairs */
   163→function buildCSV(words: CustomWord[]): string {
   164→  const lines = ['almanca,türkçe'];
   165→  for (const w of words) {
   166→    const german = w.german.includes(',') ? `"${w.german}"` : w.german;
   167→    const turkish = w.turkish.includes(',') ? `"${w.turkish}"` : w.turkish;
   168→    lines.push(`${german},${turkish}`);
   169→  }
   170→  return lines.join('\n');
   171→}

   173→/** Trigger a CSV file download */
   174→function downloadCSV(content: string, filename: string) {
   175→  const blob = new Blob([content], { type: 'text/csv;charset=utf-8;' });
   176→  const url = URL.createObjectURL(blob);
   177→  const a = document.createElement('a');
   178→  a.href = url;
   179→  a.download = filename;
   180→  document.body.appendChild(a);
   181→  a.click();
   182→  document.body.removeChild(a);
   183→  URL.revokeObjectURL(url);
   184→}

   186→function formatDate(isoString: string): string {
   187→  try {
   188→    const d = new Date(isoString);
   189→    return d.toLocaleDateString('tr-TR', { day: 'numeric', month: 'short', year: 'numeric' });
   190→  } catch {
   191→    return isoString;
   192→  }
   193→}
