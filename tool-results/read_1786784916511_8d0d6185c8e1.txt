    51→// Stable partial hints for 'high' difficulty — seeded by index so they don't re-randomize on render
    52→function makeStablePartialHints(phrases: string[]): string[] {
    53→  return phrases.map((p, i) => {
    54→    const words = p.trim().split(/\s+/);
    55→    if (words.length <= 3) return '...';
    56→    const mode = i % 3; // 0=start, 1=middle, 2=end
    57→    if (mode === 0) {
    58→      const count = Math.max(2, Math.ceil(words.length * 0.35));
    59→      return words.slice(0, count).join(' ') + ' ...';
    60→    } else if (mode === 1) {
    61→      const startIdx = Math.floor(words.length * 0.3);
    62→      const count = Math.max(2, Math.ceil(words.length * 0.4));
    63→      return '... ' + words.slice(startIdx, startIdx + count).join(' ') + ' ...';
    64→    } else {
    65→      const startIdx = Math.max(0, words.length - Math.ceil(words.length * 0.4));
    66→      return '... ' + words.slice(startIdx).join(' ');
    67→    }
    68→  });
    69→}

    76→interface SpeechRecognitionEventLike {
    77→  results: { length: number; [index: number]: SpeechRecognitionResultLike };
    78→  resultIndex: number;

    80→interface SpeechRecognitionLike {
    81→  lang: string;
    82→  continuous: boolean;
    83→  interimResults: boolean;
    84→  maxAlternatives: number;
    85→  start: () => void;
    86→  stop: () => void;

    88→  onresult: ((e: SpeechRecognitionEventLike) => void) | null;
    89→  onerror: ((e: any) => void) | null;
    90→  onend: (() => void) | null;
    91→  onstart: (() => void) | null;
    92→}

   467→    // Close hints in middle/high mode (basic keeps them visible)

   470→      setShowHints(false);

   486→      try { recognitionRef.current.abort(); } catch {}
   487→    }
   488→    tts.stop();
   489→    setPhase('ended');
   490→  }, [tts]);
   491→
   492→  // Build and send the summary when phase becomes 'ended'
   493→  useEffect(() => {
   494→    if (phase !== 'ended') return;

   496→    const tipsCount = messages.filter((m) => m.tip && m.tip.trim()).length;
   497→    const summary: ConversationSummary = {
   498→      topicId: topic.id,
   499→      topicTitleDe: topic.titleDe,
   500→      topicTitleTr: topic.titleTr,
   501→      messageCount: messages.length,
   502→      userMessageCount,
   503→      tipsCount,
   504→      durationSec: Math.round((Date.now() - startTimeRef.current) / 1000),

   512→  const replayMessage = useCallback((msg: ChatMessage) => {
   513→    tts.speak(msg.text);

   518→  const lastAiSuggestions = (lastMsg?.sender === 'ai' && lastMsg.suggestions?.length) ? lastMsg.suggestions : undefined;
   519→
   520→  // Apply hint difficulty filtering
   521→  // Priority: AI suggestions (dynamic) > static samplePhrases (fallback)
   522→  const rawHints = lastAiSuggestions || (effectiveTopic.samplePhrases.length > 0 ? effectiveTopic.samplePhrases : undefined);
