    37→interface QuizQuestion {
    38→  word: WordPair;
    39→  options: string[];
    40→  correctIndex: number;
    41→}
    42→
    43→function shuffleArray<T>(array: T[]): T[] {
    44→  const shuffled = [...array];
    45→  for (let i = shuffled.length - 1; i > 0; i--) {
    46→    const j = Math.floor(Math.random() * (i + 1));
    47→    [shuffled[i], shuffled[j]] = [shuffled[j], shuffled[i]];
    48→  }
    49→  return shuffled;
    50→}
    51→
    52→function generateQuestions(
    53→  pool: WordPair[],
    54→  count: number,
    55→  direction: Direction
    56→): QuizQuestion[] {
    57→  const shuffled = shuffleArray(pool);
    58→  const selected = shuffled.slice(0, Math.min(count, shuffled.length));
    59→
    60→  return selected.map((word) => {
    61→    const otherWords = pool.filter(
    62→      (w) =>
    63→        w.id !== word.id &&
    64→        (direction === 'de-to-tr'
    65→          ? !selected.slice(0, selected.indexOf(word) >= 0 ? selected.indexOf(word) : 0)
    66→              .some((s) => s.turkish === w.turkish)
    67→          : true)
    68→    );
    69→
    70→    const field: 'turkish' | 'german' = direction === 'de-to-tr' ? 'turkish' : 'german';
    71→    const correctValue = word[field];
    72→
    73→    // Pick 3 unique wrong answers
    74→    const usedValues = new Set<string>([correctValue]);
    75→    const wrongOptions: string[] = [];
    76→    const candidates = shuffleArray(otherWords);
    77→
    78→    for (const candidate of candidates) {
    79→      const val = candidate[field];
    80→      if (!usedValues.has(val)) {
    81→        wrongOptions.push(val);
    82→        usedValues.add(val);
    83→      }
    84→      if (wrongOptions.length >= 3) break;
    85→    }
    86→
    87→    // Fallback: if we don't have enough, allow duplicates (shouldn't happen with a decent pool)
    88→    while (wrongOptions.length < 3 && pool.length > 1) {
    89→      const rand = pool[Math.floor(Math.random() * pool.length)];
    90→      const val = rand[field];
    91→      if (val !== correctValue) {
    92→        wrongOptions.push(val);
    93→      }
    94→    }
    95→
    96→    const options = shuffleArray([correctValue, ...wrongOptions]);
    97→    const correctIndex = options.indexOf(correctValue);
    98→
    99→    return { word, options, correctIndex };
   100→  });
   101→}
